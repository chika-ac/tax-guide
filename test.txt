Sub DetectDuplicates()
    Dim wsFAQ As Worksheet
    Dim wsDup As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim j As Long
    Dim outRow As Long
    Dim qA As String
    Dim qB As String
    Dim kwA As String
    Dim kwB As String
    Dim judgment As String
    Dim matchCount As Integer
    Dim kwListA() As String
    Dim m As Long
    Dim kw As String

    Set wsFAQ = ThisWorkbook.Sheets("FAQ")

    ' Duplicatesシートを作成または初期化
    Dim sheetExists As Boolean
    sheetExists = False
    Dim s As Worksheet
    For Each s In ThisWorkbook.Sheets
        If s.Name = "Duplicates" Then
            sheetExists = True
            Exit For
        End If
    Next s

    If sheetExists Then
        Set wsDup = ThisWorkbook.Sheets("Duplicates")
        wsDup.Cells.ClearContents
        wsDup.Cells.Interior.ColorIndex = xlNone
    Else
        Set wsDup = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsDup.Name = "Duplicates"
    End If

    ' ヘッダー設定
    Dim headers(15) As String
    headers(0) = "Possibility"
    headers(1) = "Row_A"
    headers(2) = "Category_A"
    headers(3) = "Q_A"
    headers(4) = "A_A"
    headers(5) = "Keyword_A"
    headers(6) = "Row_B"
    headers(7) = "Category_B"
    headers(8) = "Q_B"
    headers(9) = "A_B"
    headers(10) = "Keyword_B"
    headers(11) = "Action"
    headers(12) = "New_Q"
    headers(13) = "New_A"
    headers(14) = "New_Keyword"
    headers(15) = "New_Shinraido"

    Dim h As Integer
    For h = 0 To 15
        wsDup.Cells(1, h + 1).Value = headers(h)
        wsDup.Cells(1, h + 1).Font.Bold = True
        wsDup.Cells(1, h + 1).Font.Name = "Arial"
        wsDup.Cells(1, h + 1).Font.Size = 9
        wsDup.Cells(1, h + 1).Interior.Color = RGB(200, 200, 200)
    Next h

    ' 列幅設定
    wsDup.Columns(1).ColumnWidth = 12
    wsDup.Columns(2).ColumnWidth = 8
    wsDup.Columns(3).ColumnWidth = 18
    wsDup.Columns(4).ColumnWidth = 35
    wsDup.Columns(5).ColumnWidth = 40
    wsDup.Columns(6).ColumnWidth = 25
    wsDup.Columns(7).ColumnWidth = 8
    wsDup.Columns(8).ColumnWidth = 18
    wsDup.Columns(9).ColumnWidth = 35
    wsDup.Columns(10).ColumnWidth = 40
    wsDup.Columns(11).ColumnWidth = 25
    wsDup.Columns(12).ColumnWidth = 18
    wsDup.Columns(13).ColumnWidth = 35
    wsDup.Columns(14).ColumnWidth = 40
    wsDup.Columns(15).ColumnWidth = 25
    wsDup.Columns(16).ColumnWidth = 12

    lastRow = wsFAQ.Cells(wsFAQ.Rows.Count, 1).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "Error: No data in FAQ sheet."
        Exit Sub
    End If

    Application.ScreenUpdating = False
    outRow = 2

    For i = 2 To lastRow - 1
        qA = Trim(wsFAQ.Cells(i, 2).Value)
        kwA = Trim(wsFAQ.Cells(i, 4).Value)
        If qA = "" Then GoTo NextI

        For j = i + 1 To lastRow
            qB = Trim(wsFAQ.Cells(j, 2).Value)
            kwB = Trim(wsFAQ.Cells(j, 4).Value)
            If qB = "" Then GoTo NextJ

            judgment = ""
            matchCount = 0

            ' Q完全一致 → 大
            If qA = qB Then
                judgment = "Large"
                GoTo WriteRow
            End If

            ' キーワード一致数カウント
            If kwA <> "" And kwB <> "" Then
                kwListA = Split(kwA, ChrW(12539))
                For m = 0 To UBound(kwListA)
                    kw = Trim(kwListA(m))
                    If Len(kw) >= 2 Then
                        If InStr(kwB, kw) > 0 Then
                            matchCount = matchCount + 1
                        End If
                    End If
                Next m
            End If

            If matchCount >= 4 Then
                judgment = "Large"
            ElseIf matchCount = 3 Then
                judgment = "Middle"
            ElseIf matchCount = 2 Then
                judgment = "Small"
            End If

            If judgment = "" Then GoTo NextJ

WriteRow:
            wsDup.Cells(outRow, 1).Value = judgment
            wsDup.Cells(outRow, 2).Value = i
            wsDup.Cells(outRow, 3).Value = wsFAQ.Cells(i, 1).Value
            wsDup.Cells(outRow, 4).Value = qA
            wsDup.Cells(outRow, 5).Value = wsFAQ.Cells(i, 3).Value
            wsDup.Cells(outRow, 6).Value = kwA
            wsDup.Cells(outRow, 7).Value = j
            wsDup.Cells(outRow, 8).Value = wsFAQ.Cells(j, 1).Value
            wsDup.Cells(outRow, 9).Value = qB
            wsDup.Cells(outRow, 10).Value = wsFAQ.Cells(j, 3).Value
            wsDup.Cells(outRow, 11).Value = kwB

            wsDup.Rows(outRow).Font.Name = "Arial"
            wsDup.Rows(outRow).Font.Size = 9
            wsDup.Cells(outRow, 5).WrapText = True
            wsDup.Cells(outRow, 10).WrapText = True
            wsDup.Rows(outRow).RowHeight = 45

            If judgment = "Large" Then
                wsDup.Cells(outRow, 1).Interior.Color = RGB(255, 199, 206)
            ElseIf judgment = "Middle" Then
                wsDup.Cells(outRow, 1).Interior.Color = RGB(255, 235, 156)
            ElseIf judgment = "Small" Then
                wsDup.Cells(outRow, 1).Interior.Color = RGB(198, 239, 206)
            End If

            On Error Resume Next
            wsDup.Cells(outRow, 12).Validation.Delete
            On Error GoTo 0
            With wsDup.Cells(outRow, 12).Validation
                .Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, _
                    Formula1:="A_Nokosu,B_Nokosu,Ryoho_Nokosu,Ryoho_Sakujo_Shinkiseisei,Ryoho_Sakujo"
                .ShowInput = True
                .ShowError = False
            End With

            outRow = outRow + 1

NextJ:
        Next j
NextI:
    Next i

    Application.ScreenUpdating = True

    MsgBox outRow - 2 & " ken no duplicate koho wo detect shimashita."
End Sub
