Sub DetectDuplicates()
    Dim wsFAQ As Worksheet
    Dim wsDup As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim j As Long
    Dim outRow As Long
    Dim qA As String
    Dim qB As String
    Dim kwA As String
    Dim kwB As String
    Dim judgment As String
    Dim found As Boolean

    Set wsFAQ = ThisWorkbook.Sheets("FAQ")

    ' 重複候補シートを作成または初期化
    On Error Resume Next
    Set wsDup = ThisWorkbook.Sheets("重複候補")
    On Error GoTo 0

    If wsDup Is Nothing Then
        Set wsDup = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsDup.Name = "Duplicates"
    Else
        wsDup.Cells.ClearContents
    End If

    ' ヘッダー
    wsDup.Cells(1, 1).Value = "FAQ_Row_A"
    wsDup.Cells(1, 2).Value = "Category_A"
    wsDup.Cells(1, 3).Value = "Q_A"
    wsDup.Cells(1, 4).Value = "FAQ_Row_B"
    wsDup.Cells(1, 5).Value = "Category_B"
    wsDup.Cells(1, 6).Value = "Q_B"
    wsDup.Cells(1, 7).Value = "Judgment"

    ' 列幅
    wsDup.Columns(1).ColumnWidth = 10
    wsDup.Columns(2).ColumnWidth = 18
    wsDup.Columns(3).ColumnWidth = 40
    wsDup.Columns(4).ColumnWidth = 10
    wsDup.Columns(5).ColumnWidth = 18
    wsDup.Columns(6).ColumnWidth = 40
    wsDup.Columns(7).ColumnWidth = 12

    ' ヘッダー書式
    wsDup.Rows(1).Font.Bold = True
    wsDup.Rows(1).Font.Name = "Arial"
    wsDup.Rows(1).Font.Size = 9

    lastRow = wsFAQ.Cells(wsFAQ.Rows.Count, 1).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "Error: No data in FAQ sheet."
        Exit Sub
    End If

    Application.ScreenUpdating = False
    outRow = 2

    ' 総当たり比較
    For i = 2 To lastRow - 1
        qA = Trim(wsFAQ.Cells(i, 2).Value)
        kwA = Trim(wsFAQ.Cells(i, 4).Value)
        If qA = "" Then GoTo NextI

        For j = i + 1 To lastRow
            qB = Trim(wsFAQ.Cells(j, 2).Value)
            kwB = Trim(wsFAQ.Cells(j, 4).Value)
            If qB = "" Then GoTo NextJ

            judgment = ""

            ' ①Qが完全一致
            If qA = qB Then
                judgment = "Kakujitsu"
                GoTo WriteRow
            End If

            ' ②Qに共通する3文字以上の文字列が含まれる
            Dim k As Long
            For k = 1 To Len(qA) - 2
                Dim substr As String
                substr = Mid(qA, k, 3)
                If InStr(qB, substr) > 0 Then
                    judgment = "Koho"
                    GoTo WriteRow
                End If
            Next k

            ' ③キーワードに共通する単語が含まれる
            If kwA <> "" And kwB <> "" Then
                Dim kwListA() As String
                Dim m As Long
                kwListA = Split(kwA, ChrW(12539))
                For m = 0 To UBound(kwListA)
                    Dim kw As String
                    kw = Trim(kwListA(m))
                    If Len(kw) >= 2 Then
                        If InStr(kwB, kw) > 0 Then
                            judgment = "Koho"
                            GoTo WriteRow
                        End If
                    End If
                Next m
            End If

            GoTo NextJ

WriteRow:
            wsDup.Cells(outRow, 1).Value = i
            wsDup.Cells(outRow, 2).Value = wsFAQ.Cells(i, 1).Value
            wsDup.Cells(outRow, 3).Value = qA
            wsDup.Cells(outRow, 4).Value = j
            wsDup.Cells(outRow, 5).Value = wsFAQ.Cells(j, 1).Value
            wsDup.Cells(outRow, 6).Value = qB
            wsDup.Cells(outRow, 7).Value = judgment
            wsDup.Rows(outRow).Font.Name = "Arial"
            wsDup.Rows(outRow).Font.Size = 9

            ' 確実な重複は赤、候補は黄
            If judgment = "Kakujitsu" Then
                wsDup.Rows(outRow).Interior.Color = RGB(255, 199, 206)
            Else
                wsDup.Rows(outRow).Interior.Color = RGB(255, 235, 156)
            End If

            outRow = outRow + 1

NextJ:
        Next j
NextI:
    Next i

    Application.ScreenUpdating = True

    MsgBox outRow - 2 & " ken no duplicate wo検出 shimashita."
End Sub
