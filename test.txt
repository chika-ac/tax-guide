Const SHEET_MANUAL  As String = "①作業手順書"
Const SHEET_ACCESS1 As String = "②Accessデータ①"
Const SHEET_ACCESS2 As String = "③Accessデータ②"
Const SHEET_SQL     As String = "④SQLデータ"
Const SHEET_CONFIG  As String = "⑤チェック設定"
Const SHEET_RESULT  As String = "⑥チェック結果"

' ============================================================
' STEP1：リセット処理
' ============================================================
Sub リセット()

    If MsgBox("前回のデータとチェック結果をすべてクリアします。よろしいですか？", _
              vbYesNo + vbQuestion, "リセット確認") = vbNo Then Exit Sub

    Dim ws1 As Worksheet
    Dim ws2 As Worksheet
    Dim ws3 As Worksheet
    Dim ws4 As Worksheet
    Dim ws6 As Worksheet
    Dim i   As Integer

    Set ws1 = ThisWorkbook.Sheets(SHEET_ACCESS1)
    Set ws2 = ThisWorkbook.Sheets(SHEET_ACCESS2)
    Set ws3 = ThisWorkbook.Sheets(SHEET_SQL)
    Set ws4 = ThisWorkbook.Sheets(SHEET_RESULT)
    Set ws6 = ThisWorkbook.Sheets(SHEET_MANUAL)

    If ws1.Cells(ws1.Rows.Count, 1).End(xlUp).Row > 2 Then
        ws1.Rows("3:" & ws1.Cells(ws1.Rows.Count, 1).End(xlUp).Row).Delete
    End If

    If ws2.Cells(ws2.Rows.Count, 1).End(xlUp).Row > 2 Then
        ws2.Rows("3:" & ws2.Cells(ws2.Rows.Count, 1).End(xlUp).Row).Delete
    End If

    If ws3.Cells(ws3.Rows.Count, 1).End(xlUp).Row >= 3 Then
        ws3.Range("A3:" & ws3.Cells(ws3.Rows.Count, 1).End(xlUp).Address).ClearContents
    End If

    If ws4.Cells(ws4.Rows.Count, 1).End(xlUp).Row > 4 Then
        ws4.Rows("5:" & ws4.Cells(ws4.Rows.Count, 1).End(xlUp).Row).Delete
    End If
    ws4.Range("D2").ClearContents
    ws4.Range("I2").ClearContents
    ws4.Range("C3").ClearContents
    ws4.Range("G3").ClearContents
    ws4.Range("I3").Value          = "全OK：未確認"
    ws4.Range("I3").Interior.Color = RGB(217, 217, 217)
    ws4.Range("I3").Font.Color     = RGB(0, 0, 0)
    ws4.Range("I3").Font.Bold      = False

    For i = 4 To 9
        ws6.Cells(i, 8).Value = "□"
    Next i

    MsgBox "リセット完了しました。STEP2から作業を開始してください。", vbInformation, "リセット完了"
    ThisWorkbook.Sheets(SHEET_MANUAL).Activate

End Sub

    ThisWorkbook.Sheets(SHEET_MANUAL).Activate

End Sub

' ============================================================
' Accessデータ読込 共通処理
' ============================================================
Private Sub Accessデータ読込(ByVal fileNo As Integer)

    Dim fileName     As String
    Dim targetSheet  As String
    Dim downloadPath As String
    Dim fullPath     As String
    Dim srcWb        As Workbook
    Dim srcWs        As Worksheet
    Dim destWs       As Worksheet
    Dim copyRange    As Range
    Dim lastRow      As Long
    Dim lastCol      As Long
    Dim r            As Long
    Dim c            As Long

    Select Case fileNo
        Case 1
            fileName    = ACCESS_FILE1
            targetSheet = SHEET_ACCESS1
        Case 2
            fileName    = ACCESS_FILE2
            targetSheet = SHEET_ACCESS2
        ' ↓ 将来ファイルが増えた場合はここに追加
        ' Case 3
        '     fileName    = ACCESS_FILE3
        '     targetSheet = "④Accessデータ③"
    End Select

    downloadPath = Environ("USERPROFILE") & "\Downloads\"
    fullPath     = downloadPath & fileName

    ' ファイルの存在確認（エラーメッセージのみ）
    If Dir(fullPath) = "" Then
        MsgBox "【エラー】ファイルが見つかりません。" & vbCrLf & vbCrLf & _
               "場所：" & fullPath & vbCrLf & vbCrLf & _
               "Accessのボタンを押してExcelを出力してから再度実行してください。", _
               vbExclamation, "ファイルなし（" & fileName & "）"
        Exit Sub
    End If

    Application.ScreenUpdating = False

    ' 対象シートをクリア（ヘッダー残す）
    Set destWs = ThisWorkbook.Sheets(targetSheet)
    If destWs.Cells(destWs.Rows.Count, 1).End(xlUp).Row > 2 Then
        destWs.Rows("3:" & destWs.Cells(destWs.Rows.Count, 1).End(xlUp).Row).Delete
    End If

    ' ソースファイルを開く
    Set srcWb = Workbooks.Open(fullPath, ReadOnly:=True)
    Set srcWs = srcWb.Sheets(1)

    lastRow = srcWs.Cells(srcWs.Rows.Count, 1).End(xlUp).Row
    lastCol = srcWs.Cells(1, srcWs.Columns.Count).End(xlToLeft).Column

    ' データなしエラー
    If lastRow < 2 Then
        srcWb.Close False
        Application.ScreenUpdating = True
        MsgBox "【エラー】データがありません。" & vbCrLf & _
               "ファイル：" & fileName, vbExclamation, "データなし"
        Exit Sub
    End If

    ' データをコピー（2行目以降）
    Set copyRange = srcWs.Range(srcWs.Cells(2, 1), srcWs.Cells(lastRow, lastCol))
    destWs.Cells(3, 1).Resize(copyRange.Rows.Count, copyRange.Columns.Count).Value = copyRange.Value

    ' 書式設定（縞模様）
    For r = 3 To 2 + copyRange.Rows.Count
        For c = 1 To lastCol
            With destWs.Cells(r, c)
                .Font.Name = "Arial"
                .Font.Size = 10
                If r Mod 2 = 0 Then
                    .Interior.Color = RGB(189, 215, 238)
                Else
                    .Interior.Color = RGB(255, 255, 255)
                End If
            End With
        Next c
    Next r

    srcWb.Close False
    Application.ScreenUpdating = True

End Sub

' ============================================================
' STEP4：チェック実行メイン
' ============================================================
Sub チェック実行()

    If MsgBox("チェックを実行します。データの取込は完了していますか？", _
              vbYesNo + vbQuestion, "チェック実行確認") = vbNo Then Exit Sub

    Application.ScreenUpdating = False

    Dim wsResult As Worksheet
    Dim wsConfig As Worksheet
    Dim wsTarget As Worksheet

    Set wsResult = ThisWorkbook.Sheets(SHEET_RESULT)
    Set wsConfig = ThisWorkbook.Sheets(SHEET_CONFIG)

    If wsResult.Cells(wsResult.Rows.Count, 1).End(xlUp).Row > 4 Then
        wsResult.Rows("5:" & wsResult.Cells(wsResult.Rows.Count, 1).End(xlUp).Row).Delete
    End If

    wsResult.Range("D2").Value        = Now()
    wsResult.Range("D2").NumberFormat = "yyyy/mm/dd hh:mm:ss"

    Dim configLastRow   As Long
    configLastRow = wsConfig.Cells(wsConfig.Rows.Count, 1).End(xlUp).Row

    ' 変数宣言
    Dim resultRow1      As Long
    Dim resultRow2      As Long
    Dim totalNG         As Long
    Dim totalCheck      As Long
    Dim i               As Long
    Dim j               As Long
    Dim k               As Long
    Dim col             As Long
    Dim dataKind        As String
    Dim colName         As String
    Dim checkType       As String
    Dim cond1           As String
    Dim cond2           As String
    Dim confirmNote     As String
    Dim targetSheetName As String
    Dim targetCol       As Long
    Dim headerRow       As Long
    Dim dataStartRow    As Long
    Dim dataLastRow     As Long
    Dim lastColTarget   As Long
    Dim totalRows       As Long
    Dim ngCount         As Long
    Dim cellVal         As String
    Dim numVal          As Double
    Dim outOfRange      As Boolean
    Dim strLen          As Long
    Dim lenNG           As Boolean
    Dim dupDict         As Object
    Dim key             As Variant
    Dim ngValues        As Object
    Dim ngVal           As Variant
    Dim isFirst         As Boolean

    totalNG    = 0
    totalCheck = 0

    ' ============================================================
    ' Accessデータ① NG一覧セクション描画
    ' ============================================================
    wsResult.Rows(4).RowHeight = 8

    Dim sec1HeaderRow As Long
    sec1HeaderRow = 5

    Call セクションタイトル描画(wsResult, sec1HeaderRow, _
                                "▼ Accessデータ① NG一覧", 11, RGB(31, 78, 121))

    Dim sec1ColHeaders(11) As String
    sec1ColHeaders(0)  = "#"
    sec1ColHeaders(1)  = "列名"
    sec1ColHeaders(2)  = "チェック種別"
    sec1ColHeaders(3)  = "件数"
    sec1ColHeaders(4)  = "NG数"
    sec1ColHeaders(5)  = "判定"
    sec1ColHeaders(6)  = "申請番号"
    sec1ColHeaders(7)  = "職員番号"
    sec1ColHeaders(8)  = "氏名"
    sec1ColHeaders(9)  = "NG値"
    sec1ColHeaders(10) = "確認手順"
    sec1ColHeaders(11) = "済"

    Call ヘッダー行描画(wsResult, sec1HeaderRow + 1, sec1ColHeaders, 11, RGB(31, 78, 121))

    resultRow1 = sec1HeaderRow + 2

    ' ============================================================
    ' チェック設定ループ（Accessデータ①）
    ' ============================================================
    For i = 4 To configLastRow

        dataKind    = wsConfig.Cells(i, 1).Value
        colName     = wsConfig.Cells(i, 2).Value
        checkType   = wsConfig.Cells(i, 3).Value
        cond1       = wsConfig.Cells(i, 4).Value
        cond2       = wsConfig.Cells(i, 5).Value
        confirmNote = wsConfig.Cells(i, 8).Value

        If dataKind = "" Or colName = "" Then GoTo NextConfig
        If dataKind <> "Accessデータ①" Then GoTo NextConfig

        targetSheetName = SHEET_ACCESS1
        Set wsTarget    = ThisWorkbook.Sheets(targetSheetName)
        headerRow       = 2

        targetCol     = 0
        lastColTarget = wsTarget.Cells(headerRow, wsTarget.Columns.Count).End(xlToLeft).Column

        For j = 1 To lastColTarget
            If wsTarget.Cells(headerRow, j).Value = colName Then
                targetCol = j
                Exit For
            End If
        Next j

        If targetCol = 0 Then GoTo NextConfig

        dataStartRow = headerRow + 1
        dataLastRow  = wsTarget.Cells(wsTarget.Rows.Count, 1).End(xlUp).Row

        If dataLastRow < dataStartRow Then GoTo NextConfig

        totalRows = dataLastRow - dataStartRow + 1
        ngCount   = 0
        Set ngValues = CreateObject("System.Collections.ArrayList")

        Select Case checkType

            Case "空白チェック"
                For k = dataStartRow To dataLastRow
                    If wsTarget.Cells(k, targetCol).Value = "" Then
                        ngCount = ngCount + 1
                        ngValues.Add Array(wsTarget.Cells(k, 1).Value, _
                                          wsTarget.Cells(k, 2).Value, _
                                          wsTarget.Cells(k, 3).Value, _
                                          "(空白)")
                    End If
                Next k

            Case "重複チェック"
                Set dupDict = CreateObject("Scripting.Dictionary")
                For k = dataStartRow To dataLastRow
                    cellVal = CStr(wsTarget.Cells(k, targetCol).Value)
                    If cellVal <> "" Then
                        If dupDict.Exists(cellVal) Then
                            dupDict(cellVal) = dupDict(cellVal) + 1
                        Else
                            dupDict.Add cellVal, 1
                        End If
                    End If
                Next k
                For Each key In dupDict.Keys
                    If dupDict(key) > 1 Then
                        ngCount = ngCount + 1
                        ngValues.Add Array("", "", "", _
                                          """" & key & """ (" & dupDict(key) & "件重複)")
                    End If
                Next key

            Case "範囲チェック"
                If cond1 = "" And cond2 = "" Then GoTo NextConfig
                For k = dataStartRow To dataLastRow
                    If IsNumeric(wsTarget.Cells(k, targetCol).Value) Then
                        numVal     = CDbl(wsTarget.Cells(k, targetCol).Value)
                        outOfRange = False
                        If cond1 <> "" And numVal < CDbl(cond1) Then outOfRange = True
                        If cond2 <> "" And numVal > CDbl(cond2) Then outOfRange = True
                        If outOfRange Then
                            ngCount = ngCount + 1
                            ngValues.Add Array(wsTarget.Cells(k, 1).Value, _
                                              wsTarget.Cells(k, 2).Value, _
                                              wsTarget.Cells(k, 3).Value, _
                                              CStr(numVal))
                        End If
                    End If
                Next k

            Case "ワードチェック"
                If cond1 = "" Then GoTo NextConfig
                For k = dataStartRow To dataLastRow
                    If InStr(CStr(wsTarget.Cells(k, targetCol).Value), cond1) > 0 Then
                        ngCount = ngCount + 1
                        ngValues.Add Array(wsTarget.Cells(k, 1).Value, _
                                          wsTarget.Cells(k, 2).Value, _
                                          wsTarget.Cells(k, 3).Value, _
                                          CStr(wsTarget.Cells(k, targetCol).Value))
                    End If
                Next k

            Case "文字数チェック"
                If cond1 = "" And cond2 = "" Then GoTo NextConfig
                For k = dataStartRow To dataLastRow
                    strLen = Len(CStr(wsTarget.Cells(k, targetCol).Value))
                    lenNG  = False
                    If cond1 <> "" And strLen < CLng(cond1) Then lenNG = True
                    If cond2 <> "" And strLen > CLng(cond2) Then lenNG = True
                    If lenNG Then
                        ngCount = ngCount + 1
                        ngValues.Add Array(wsTarget.Cells(k, 1).Value, _
                                          wsTarget.Cells(k, 2).Value, _
                                          wsTarget.Cells(k, 3).Value, _
                                          CStr(wsTarget.Cells(k, targetCol).Value) & _
                                          " (" & strLen & "文字)")
                    End If
                Next k

        End Select

        totalCheck = totalCheck + 1
        If ngCount = 0 Then GoTo NextConfig

        totalNG = totalNG + 1
        isFirst = True

        For Each ngVal In ngValues
            With wsResult
                If isFirst Then
                    .Cells(resultRow1, 1).Value  = totalCheck
                    .Cells(resultRow1, 2).Value  = colName
                    .Cells(resultRow1, 3).Value  = checkType
                    .Cells(resultRow1, 4).Value  = totalRows & "件"
                    .Cells(resultRow1, 5).Value  = ngCount
                    .Cells(resultRow1, 6).Value  = "❌NG"
                    isFirst = False
                Else
                    .Cells(resultRow1, 1).Value  = ""
                    .Cells(resultRow1, 2).Value  = ""
                    .Cells(resultRow1, 3).Value  = ""
                    .Cells(resultRow1, 4).Value  = ""
                    .Cells(resultRow1, 5).Value  = ""
                    .Cells(resultRow1, 6).Value  = ""
                End If

                .Cells(resultRow1, 7).Value  = ngVal(0)
                .Cells(resultRow1, 8).Value  = ngVal(1)
                .Cells(resultRow1, 9).Value  = ngVal(2)
                .Cells(resultRow1, 10).Value = ngVal(3)
                .Cells(resultRow1, 11).Value = confirmNote
                .Cells(resultRow1, 12).Value = "□"

                Call NG行書式設定(wsResult, resultRow1, 12)
                .Cells(resultRow1, 6).Font.Color = RGB(192, 0, 0)
                .Cells(resultRow1, 6).Font.Bold  = True
                .Cells(resultRow1, 11).WrapText  = True
            End With
            resultRow1 = resultRow1 + 1
        Next ngVal

        Call 区切り線描画(wsResult, resultRow1 - 1, 12)

NextConfig:
    Next i

    ' ============================================================
    ' Accessデータ②・SQL NG一覧セクション描画
    ' ============================================================
    wsResult.Rows(resultRow1).RowHeight = 8

    Dim sec2StartRow As Long
    sec2StartRow = resultRow1 + 1

    Call セクションタイトル描画(wsResult, sec2StartRow, _
                                "▼ Accessデータ②・SQL NG一覧", 11, RGB(31, 78, 121))

    Dim sec2ColHeaders(9) As String
    sec2ColHeaders(0) = "#"
    sec2ColHeaders(1) = "データ種別"
    sec2ColHeaders(2) = "列名"
    sec2ColHeaders(3) = "チェック種別"
    sec2ColHeaders(4) = "件数"
    sec2ColHeaders(5) = "NG数"
    sec2ColHeaders(6) = "判定"
    sec2ColHeaders(7) = "NG値"
    sec2ColHeaders(8) = "確認手順"
    sec2ColHeaders(9) = "済"

    Call ヘッダー行描画(wsResult, sec2StartRow + 1, sec2ColHeaders, 9, RGB(31, 78, 121))

    resultRow2 = sec2StartRow + 2

    ' ============================================================
    ' チェック設定ループ（Accessデータ②・SQL）
    ' ============================================================
    Dim totalCheck2 As Long
    totalCheck2 = 0

    For i = 4 To configLastRow

        dataKind    = wsConfig.Cells(i, 1).Value
        colName     = wsConfig.Cells(i, 2).Value
        checkType   = wsConfig.Cells(i, 3).Value
        cond1       = wsConfig.Cells(i, 4).Value
        cond2       = wsConfig.Cells(i, 5).Value
        confirmNote = wsConfig.Cells(i, 8).Value

        If dataKind = "" Or colName = "" Then GoTo NextConfig2
        If dataKind = "Accessデータ①" Then GoTo NextConfig2

        Select Case dataKind
            Case "Accessデータ②": targetSheetName = SHEET_ACCESS2
            Case "SQLデータ":     targetSheetName = SHEET_SQL
            Case Else:             GoTo NextConfig2
        End Select

        Set wsTarget = ThisWorkbook.Sheets(targetSheetName)

        If targetSheetName = SHEET_SQL Then
            headerRow = 3
        Else
            headerRow = 2
        End If

        targetCol     = 0
        lastColTarget = wsTarget.Cells(headerRow, wsTarget.Columns.Count).End(xlToLeft).Column

        For j = 1 To lastColTarget
            If wsTarget.Cells(headerRow, j).Value = colName Then
                targetCol = j
                Exit For
            End If
        Next j

        If targetCol = 0 Then GoTo NextConfig2

        dataStartRow = headerRow + 1
        dataLastRow  = wsTarget.Cells(wsTarget.Rows.Count, 1).End(xlUp).Row

        If dataLastRow < dataStartRow Then GoTo NextConfig2

        totalRows = dataLastRow - dataStartRow + 1
        ngCount   = 0
        Set ngValues = CreateObject("System.Collections.ArrayList")

        Select Case checkType

            Case "空白チェック"
                For k = dataStartRow To dataLastRow
                    If wsTarget.Cells(k, targetCol).Value = "" Then
                        ngCount = ngCount + 1
                        ngValues.Add "行" & k & "：(空白)"
                    End If
                Next k

            Case "重複チェック"
                Set dupDict = CreateObject("Scripting.Dictionary")
                For k = dataStartRow To dataLastRow
                    cellVal = CStr(wsTarget.Cells(k, targetCol).Value)
                    If cellVal <> "" Then
                        If dupDict.Exists(cellVal) Then
                            dupDict(cellVal) = dupDict(cellVal) + 1
                        Else
                            dupDict.Add cellVal, 1
                        End If
                    End If
                Next k
                For Each key In dupDict.Keys
                    If dupDict(key) > 1 Then
                        ngCount = ngCount + 1
                        ngValues.Add """" & key & """ (" & dupDict(key) & "件重複)"
                    End If
                Next key

            Case "範囲チェック"
                If cond1 = "" And cond2 = "" Then GoTo NextConfig2
                For k = dataStartRow To dataLastRow
                    If IsNumeric(wsTarget.Cells(k, targetCol).Value) Then
                        numVal     = CDbl(wsTarget.Cells(k, targetCol).Value)
                        outOfRange = False
                        If cond1 <> "" And numVal < CDbl(cond1) Then outOfRange = True
                        If cond2 <> "" And numVal > CDbl(cond2) Then outOfRange = True
                        If outOfRange Then
                            ngCount = ngCount + 1
                            ngValues.Add "行" & k & "：" & CStr(numVal)
                        End If
                    End If
                Next k

            Case "ワードチェック"
                If cond1 = "" Then GoTo NextConfig2
                For k = dataStartRow To dataLastRow
                    If InStr(CStr(wsTarget.Cells(k, targetCol).Value), cond1) > 0 Then
                        ngCount = ngCount + 1
                        ngValues.Add "行" & k & "：" & CStr(wsTarget.Cells(k, targetCol).Value)
                    End If
                Next k

            Case "文字数チェック"
                If cond1 = "" And cond2 = "" Then GoTo NextConfig2
                For k = dataStartRow To dataLastRow
                    strLen = Len(CStr(wsTarget.Cells(k, targetCol).Value))
                    lenNG  = False
                    If cond1 <> "" And strLen < CLng(cond1) Then lenNG = True
                    If cond2 <> "" And strLen > CLng(cond2) Then lenNG = True
                    If lenNG Then
                        ngCount = ngCount + 1
                        ngValues.Add "行" & k & "：" & _
                                     CStr(wsTarget.Cells(k, targetCol).Value) & _
                                     " (" & strLen & "文字)"
                    End If
                Next k

        End Select

        totalCheck2 = totalCheck2 + 1
        If ngCount = 0 Then GoTo NextConfig2

        totalNG = totalNG + 1
        isFirst = True

        For Each ngVal In ngValues
            With wsResult
                If isFirst Then
                    .Cells(resultRow2, 1).Value = totalCheck2
                    .Cells(resultRow2, 2).Value = dataKind
                    .Cells(resultRow2, 3).Value = colName
                    .Cells(resultRow2, 4).Value = checkType
                    .Cells(resultRow2, 5).Value = totalRows & "件"
                    .Cells(resultRow2, 6).Value = ngCount
                    .Cells(resultRow2, 7).Value = "❌NG"
                    isFirst = False
                Else
                    .Cells(resultRow2, 1).Value = ""
                    .Cells(resultRow2, 2).Value = ""
                    .Cells(resultRow2, 3).Value = ""
                    .Cells(resultRow2, 4).Value = ""
                    .Cells(resultRow2, 5).Value = ""
                    .Cells(resultRow2, 6).Value = ""
                    .Cells(resultRow2, 7).Value = ""
                End If

                .Cells(resultRow2, 8).Value  = CStr(ngVal)
                .Cells(resultRow2, 9).Value  = confirmNote
                .Cells(resultRow2, 10).Value = "□"

                Call NG行書式設定(wsResult, resultRow2, 10)
                .Cells(resultRow2, 7).Font.Color = RGB(192, 0, 0)
                .Cells(resultRow2, 7).Font.Bold  = True
                .Cells(resultRow2, 9).WrapText   = True
            End With
            resultRow2 = resultRow2 + 1
        Next ngVal

        Call 区切り線描画(wsResult, resultRow2 - 1, 10)

NextConfig2:
    Next i

    ' ============================================================
    ' Accessデータ②専用処理（I列チェック＋全行引用）
    ' ============================================================
    Call Access2専用処理(wsResult, resultRow2, totalNG, wsConfig)

    ' ============================================================
    ' サマリー更新
    ' ============================================================
    wsResult.Range("C3").Value = totalCheck & "件"
    wsResult.Range("G3").Value = totalNG & "件"

    If totalNG = 0 Then
        wsResult.Range("I3").Value          = "✅全チェックOK"
        wsResult.Range("I3").Interior.Color = RGB(198, 239, 206)
        wsResult.Range("I3").Font.Color     = RGB(55, 86, 35)
        wsResult.Range("I3").Font.Bold      = True
    Else
        wsResult.Range("I3").Value          = "❌NG " & totalNG & "件あり"
        wsResult.Range("I3").Interior.Color = RGB(255, 199, 206)
        wsResult.Range("I3").Font.Color     = RGB(192, 0, 0)
        wsResult.Range("I3").Font.Bold      = True
    End If

    Application.ScreenUpdating = True

    ThisWorkbook.Sheets(SHEET_MANUAL).Cells(8, 8).Value = "✅"

    wsResult.Activate
    wsResult.Range("A1").Select

    If totalNG = 0 Then
        MsgBox "チェック完了！" & vbCrLf & "NGはありませんでした。", _
               vbInformation, "チェック完了"
    Else
        MsgBox "チェック完了！" & vbCrLf & "NG：" & totalNG & "件あります。" & vbCrLf & vbCrLf & _
               "赤くなっている行を確認し、対応後に済チェックを入れてください。", _
               vbExclamation, "チェック完了（NG有）"
    End If

End Sub

' ============================================================
' Accessデータ②専用処理
' ============================================================
Private Sub Access2専用処理(ByRef wsResult As Worksheet, _
                             ByRef resultRow As Long, _
                             ByRef totalNG As Long, _
                             ByVal wsConfig As Worksheet)

    Dim wsAccess2    As Worksheet
    Dim dataLastRow  As Long
    Dim dataStartRow As Long
    Dim k            As Long
    Dim col          As Long
    Dim colIdx       As Long
    Dim iVal         As Variant
    Dim isNG         As Boolean
    Dim ngCount      As Long
    Dim ngValues     As Object
    Dim ngVal        As Variant
    Dim isFirst      As Boolean
    Dim confirmMemo  As String
    Dim memoLines()  As String
    Dim memoLine     As Variant
    Dim memoRow      As Long

    Dim quoteCols(7) As Long
    quoteCols(0) = 1
    quoteCols(1) = 2
    quoteCols(2) = 3
    quoteCols(3) = 5
    quoteCols(4) = 6
    quoteCols(5) = 7
    quoteCols(6) = 8
    quoteCols(7) = 10

    Set wsAccess2 = ThisWorkbook.Sheets(SHEET_ACCESS2)
    dataStartRow  = 3
    dataLastRow   = wsAccess2.Cells(wsAccess2.Rows.Count, 1).End(xlUp).Row

    If dataLastRow < dataStartRow Then Exit Sub

    ' ============================================================
    ' I列チェック
    ' ============================================================
    ngCount = 0
    Set ngValues = CreateObject("System.Collections.ArrayList")

    For k = dataStartRow To dataLastRow
        iVal = wsAccess2.Cells(k, 9).Value
        isNG = False
        If iVal = "" Then
            isNG = True
        ElseIf Not IsNumeric(iVal) Then
            isNG = True
        ElseIf CDbl(iVal) <> 1 Then
            isNG = True
        End If

        If isNG Then
            ngCount = ngCount + 1
            ngValues.Add Array(wsAccess2.Cells(k, 1).Value, _
                               wsAccess2.Cells(k, 2).Value, _
                               wsAccess2.Cells(k, 3).Value, _
                               IIf(iVal = "", "(空白)", CStr(iVal)))
        End If
    Next k

    If ngCount > 0 Then
        totalNG = totalNG + 1
        isFirst = True

        For Each ngVal In ngValues
            With wsResult
                If isFirst Then
                    .Cells(resultRow, 1).Value = ngCount
                    .Cells(resultRow, 2).Value = "Accessデータ②"
                    .Cells(resultRow, 3).Value = "I列"
                    .Cells(resultRow, 4).Value = "I列チェック(1以外NG)"
                    .Cells(resultRow, 5).Value = (dataLastRow - dataStartRow + 1) & "件"
                    .Cells(resultRow, 6).Value = ngCount
                    .Cells(resultRow, 7).Value = "❌NG"
                    isFirst = False
                Else
                    .Cells(resultRow, 1).Value = ""
                    .Cells(resultRow, 2).Value = ""
                    .Cells(resultRow, 3).Value = ""
                    .Cells(resultRow, 4).Value = ""
                    .Cells(resultRow, 5).Value = ""
                    .Cells(resultRow, 6).Value = ""
                    .Cells(resultRow, 7).Value = ""
                End If

                .Cells(resultRow, 8).Value  = "I列=" & ngVal(3) & _
                                              " 申請：" & ngVal(0) & _
                                              " 職員番号：" & ngVal(1) & _
                                              " 氏名：" & ngVal(2)
                .Cells(resultRow, 9).Value  = ""
                .Cells(resultRow, 10).Value = "□"

                Call NG行書式設定(wsResult, resultRow, 10)
                .Cells(resultRow, 7).Font.Color = RGB(192, 0, 0)
                .Cells(resultRow, 7).Font.Bold  = True
            End With
            resultRow = resultRow + 1
        Next ngVal

        Call 区切り線描画(wsResult, resultRow - 1, 10)
    End If

    ' ============================================================
    ' 全行引用セクション
    ' ============================================================
    wsResult.Rows(resultRow).RowHeight = 8
    resultRow = resultRow + 1

    ' チェック設定シートから目視確認ポイントを取得
    confirmMemo = ""
    For memoRow = 1 To wsConfig.Rows.Count
        If InStr(wsConfig.Cells(memoRow, 1).Value, "Accessデータ②目視確認ポイント") > 0 Then
            confirmMemo = wsConfig.Cells(memoRow + 1, 1).Value
            Exit For
        End If
    Next memoRow

    Call セクションタイトル描画(wsResult, resultRow, _
                                "▼ Accessデータ② 目視確認用引用一覧", 11, RGB(31, 78, 121))
    resultRow = resultRow + 1

    ' 確認ポイントを改行ごとに1行ずつ展開
    If confirmMemo <> "" Then
        With wsResult
            .Cells(resultRow, 1).Value      = "【確認ポイント】"
            .Cells(resultRow, 1).Font.Name  = "Arial"
            .Cells(resultRow, 1).Font.Size  = 10
            .Cells(resultRow, 1).Font.Bold  = True
            .Cells(resultRow, 1).Font.Color = RGB(192, 0, 0)
            With .Range(.Cells(resultRow, 1), .Cells(resultRow, QUOTE_COL_COUNT + 1))
                .Interior.Color = RGB(255, 242, 204)
            End With
            .Rows(resultRow).RowHeight = 18
        End With
        resultRow = resultRow + 1

        memoLines = Split(confirmMemo, vbLf)
        For Each memoLine In memoLines
            If Trim(CStr(memoLine)) <> "" Then
                With wsResult
                    .Cells(resultRow, 1).Value = Trim(CStr(memoLine))
                    .Cells(resultRow, 1).Font.Name = "Arial"
                    .Cells(resultRow, 1).Font.Size = 10
                    With .Range(.Cells(resultRow, 1), .Cells(resultRow, QUOTE_COL_COUNT + 1))
                        .Merge
                        .Interior.Color = RGB(255, 242, 204)
                    End With
                    .Rows(resultRow).RowHeight = 18
                End With
                resultRow = resultRow + 1
            End If
        Next memoLine
    End If

    ' 引用一覧ヘッダー
    Dim quoteHeaders(8) As String
    For colIdx = 0 To 7
        quoteHeaders(colIdx) = wsAccess2.Cells(2, quoteCols(colIdx)).Value
    Next colIdx
    quoteHeaders(8) = "済"

    Call ヘッダー行描画(wsResult, resultRow, quoteHeaders, 8, RGB(31, 78, 121))
    resultRow = resultRow + 1

    ' データ全行出力
    Dim quoteRowIdx As Long
    quoteRowIdx = 0
    For k = dataStartRow To dataLastRow
        For colIdx = 0 To 7
            With wsResult.Cells(resultRow, colIdx + 1)
                .Value             = wsAccess2.Cells(k, quoteCols(colIdx)).Value
                .Font.Name         = "Arial"
                .Font.Size         = 10
                .VerticalAlignment = xlVAlignCenter
                If quoteRowIdx Mod 2 = 0 Then
                    .Interior.Color = RGB(189, 215, 238)
                Else
                    .Interior.Color = RGB(255, 255, 255)
                End If
            End With
        Next colIdx

        With wsResult.Cells(resultRow, QUOTE_COL_COUNT + 1)
            .Value               = "□"
            .Font.Name           = "Arial"
            .Font.Size           = 10
            .HorizontalAlignment = xlHAlignCenter
            .VerticalAlignment   = xlVAlignCenter
            If quoteRowIdx Mod 2 = 0 Then
                .Interior.Color = RGB(189, 215, 238)
            Else
                .Interior.Color = RGB(255, 255, 255)
            End If
        End With

        With wsResult.Range(wsResult.Cells(resultRow, 1), _
                            wsResult.Cells(resultRow, QUOTE_COL_COUNT + 1)).Borders
            .LineStyle = xlContinuous
            .Color     = RGB(191, 191, 191)
            .Weight    = xlThin
        End With

        quoteRowIdx = quoteRowIdx + 1
        resultRow   = resultRow + 1
    Next k

End Sub

' ============================================================
' 共通：セクションタイトル行描画
' ============================================================
Private Sub セクションタイトル描画(ByVal ws As Worksheet, _
                                    ByVal rowNum As Long, _
                                    ByVal title As String, _
                                    ByVal fontSize As Integer, _
                                    ByVal bgColor As Long)
    With ws
        .Cells(rowNum, 1).Value = title
        With .Range(.Cells(rowNum, 1), .Cells(rowNum, 12))
            .Merge
            .Font.Name           = "Arial"
            .Font.Size           = fontSize
            .Font.Bold           = True
            .Font.Color          = RGB(255, 255, 255)
            .Interior.Color      = bgColor
            .VerticalAlignment   = xlVAlignCenter
            .HorizontalAlignment = xlHAlignLeft
        End With
        .Rows(rowNum).RowHeight = 22
    End With
End Sub

' ============================================================
' 共通：ヘッダー行描画
' ============================================================
Private Sub ヘッダー行描画(ByVal ws As Worksheet, _
                            ByVal rowNum As Long, _
                            ByVal headers As Variant, _
                            ByVal colCount As Long, _
                            ByVal bgColor As Long)
    Dim c As Long
    For c = 0 To colCount
        With ws.Cells(rowNum, c + 1)
            .Value               = headers(c)
            .Font.Name           = "Arial"
            .Font.Size           = 10
            .Font.Bold           = True
            .Font.Color          = RGB(255, 255, 255)
            .Interior.Color      = bgColor
            .VerticalAlignment   = xlVAlignCenter
            .HorizontalAlignment = xlHAlignCenter
        End With
    Next c

    With ws.Range(ws.Cells(rowNum, 1), ws.Cells(rowNum, colCount + 1)).Borders
        .LineStyle = xlContinuous
        .Color     = RGB(191, 191, 191)
        .Weight    = xlThin
    End With

    ws.Rows(rowNum).RowHeight = 22
End Sub

' ============================================================
' 共通：NG行書式設定
' ============================================================
Private Sub NG行書式設定(ByVal ws As Worksheet, _
                          ByVal rowNum As Long, _
                          ByVal colCount As Long)
    Dim c As Long
    For c = 1 To colCount
        With ws.Cells(rowNum, c)
            .Font.Name         = "Arial"
            .Font.Size         = 10
            .VerticalAlignment = xlVAlignCenter
            .Interior.Color    = RGB(255, 199, 206)
            If c = 1 Or c = 5 Or c = 6 Or c = colCount Then
                .HorizontalAlignment = xlHAlignCenter
            End If
        End With
    Next c

    With ws.Range(ws.Cells(rowNum, 1), ws.Cells(rowNum, colCount)).Borders
        .LineStyle = xlContinuous
        .Color     = RGB(191, 191, 191)
        .Weight    = xlThin
    End With
End Sub

' ============================================================
' 共通：区切り線描画
' ============================================================
Private Sub 区切り線描画(ByVal ws As Worksheet, _
                          ByVal rowNum As Long, _
                          ByVal colCount As Long)
    With ws.Range(ws.Cells(rowNum, 1), _
                  ws.Cells(rowNum, colCount)).Borders(xlEdgeBottom)
        .LineStyle = xlContinuous
        .Color     = RGB(192, 0, 0)
        .Weight    = xlMedium
    End With
End Sub
