Sub CheckData()
    Dim r As Long
    Dim result As String
    For r = 2 To 100
        If InStr(ThisWorkbook.Sheets("データ").Cells(r, 10).Value, "決裁") > 0 Then
            result = result & r & ": [" & ThisWorkbook.Sheets("データ").Cells(r, 10).Value & "]" & vbCrLf
        End If
    Next r
    MsgBox result
End Sub
