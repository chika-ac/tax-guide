Sub ExportFinalFAQtoExcel()
    Dim wsFinal As Worksheet
    Dim fPath As Variant
    Dim newWb As Workbook
    Dim lastRow As Long

    ' FinalFAQシートの確認
    Dim sheetExists As Boolean
    sheetExists = False
    Dim s As Worksheet
    For Each s In ThisWorkbook.Sheets
        If s.Name = "FinalFAQ" Then
            sheetExists = True
            Exit For
        End If
    Next s

    If Not sheetExists Then
        MsgBox "Error: FinalFAQ sheet not found. Please run ExportFinalFAQ first."
        Exit Sub
    End If

    Set wsFinal = ThisWorkbook.Sheets("FinalFAQ")

    lastRow = wsFinal.Cells(wsFinal.Rows.Count, 1).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "Error: No data in FinalFAQ sheet."
        Exit Sub
    End If

    ' 保存先を選択
    fPath = Application.GetSaveAsFilename( _
        InitialFileName:="FinalFAQ_" & Format(Now, "YYYYMMDD") & ".xlsx", _
        FileFilter:="Excel," & Chr(42) & ".xlsx")
    If VarType(fPath) = vbBoolean Then Exit Sub

    ' 新しいブックを作成してコピー
    Set newWb = Workbooks.Add
    wsFinal.Copy Before:=newWb.Sheets(1)

    ' デフォルトシートを削除
    Application.DisplayAlerts = False
    newWb.Sheets(newWb.Sheets.Count).Delete
    Application.DisplayAlerts = True

    ' 保存して閉じる
    newWb.SaveAs Filename:=fPath, FileFormat:=xlOpenXMLWorkbook
    newWb.Close

    MsgBox "FinalFAQ wo Excel ni hozon shimashita."
End Sub
