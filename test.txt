Sub Macro1()
    On Error GoTo ErrorHandler

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False

    Dim wsShomu As Worksheet, wsIPK As Worksheet, wsDebug As Worksheet
    Dim lastRowShomu As Long, lastRowIPK As Long
    Dim i As Long
    Dim result As String
    Dim dict As Object
    Dim key As Variant
    Dim status As String
    Dim accountType As String
    Dim ipkDataBackup As Variant

    ' ヘッダー文字列
    Dim HEADER_KOUZA_ADD As String
    Dim HEADER_FURIKAE As String
    HEADER_KOUZA_ADD = "【IPK突合判定結果】【口座追加あり】"
    HEADER_FURIKAE = "【IPK突合判定結果】【振替額変更のみ】"

    Set wsShomu = ThisWorkbook.Sheets("庶務事務")
    Set wsIPK = ThisWorkbook.Sheets("IPK")

    ' IPKシートのデータをバックアップ
    ipkDataBackup = wsIPK.UsedRange.Value
    
    ' デバッグ用シートの設定
    On Error Resume Next
    Set wsDebug = ThisWorkbook.Sheets("Debug")
    If wsDebug Is Nothing Then
        Set wsDebug = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsDebug.Name = "Debug"
    End If
    On Error GoTo ErrorHandler
    wsDebug.Cells.ClearContents

    ' 1. IPKシートY～AE列のデータをクリアしてから削除
    Application.StatusBar = "1. IPKシートY～AE列のデータをクリアしてから削除"
    wsIPK.Range("Y:AE").ClearContents
    wsIPK.Range("Y:AE").Delete

    ' 2. IPKシートW列のデータをクリアしてから削除
    Application.StatusBar = "2. IPKシートW列のデータをクリアしてから削除"
    wsIPK.Columns("W").ClearContents
    wsIPK.Columns("W").Delete

    ' 3. IPKシートU列のデータをクリアしてから削除
    Application.StatusBar = "3. IPKシートU列のデータをクリアしてから削除"
    wsIPK.Columns("U").ClearContents
    wsIPK.Columns("U").Delete

    ' 4. IPKシートD～S列のデータをクリアしてから削除
    Application.StatusBar = "4. IPKシートD～S列のデータをクリアしてから削除"
    wsIPK.Range("D:S").ClearContents
    wsIPK.Range("D:S").Delete

    ' 5. IPKシートB列のデータをクリアしてから削除
    Application.StatusBar = "5. IPKシートB列のデータをクリアしてから削除"
    wsIPK.Columns("B").ClearContents
    wsIPK.Columns("B").Delete

    ' 6. IPKシートの不要な記号を空白に置換する
    Application.StatusBar = "6. IPKシートの不要な記号を空白に置換する"
    Dim unicodeChar As String
    unicodeChar = ChrW(&H201D)
    With wsIPK.Range("A:AE")
        .Replace What:=unicodeChar, Replacement:="", LookAt:=xlPart, _
                 SearchOrder:=xlByRows, MatchCase:=False
    End With

    ' BCDE列を数値形式に変換
    wsIPK.Columns("B:E").NumberFormat = "0"

    ' 7. 条件に基づいて庶務事務シートQ列に記述
    Application.StatusBar = "7. 条件に基づいて庶務事務シートQ列に記述"
    lastRowShomu = wsShomu.Cells(wsShomu.Rows.Count, "B").End(xlUp).Row
    lastRowIPK = wsIPK.Cells(wsIPK.Rows.Count, "A").End(xlUp).Row

    Dim shomuData As Variant, ipkData As Variant
    shomuData = wsShomu.Range("A2:J" & lastRowShomu).Value
    ipkData = wsIPK.Range("A2:E" & lastRowIPK).Value

    ' デバッグ用シートにヘッダーを追加
    wsDebug.Cells(1, 1).Value = "Shomu_B"
    wsDebug.Cells(1, 2).Value = "Shomu_E"
    wsDebug.Cells(1, 3).Value = "Shomu_G"
    wsDebug.Cells(1, 4).Value = "Shomu_I"
    wsDebug.Cells(1, 5).Value = "Shomu_J"
    wsDebug.Cells(1, 6).Value = "IPK_B"
    wsDebug.Cells(1, 7).Value = "IPK_C"
    wsDebug.Cells(1, 8).Value = "IPK_D"
    wsDebug.Cells(1, 9).Value = "IPK_E"
    wsDebug.Cells(1, 10).Value = "Shomu_Key"
    wsDebug.Cells(1, 11).Value = "IPK_Key"

    ' 辞書を作成してIPKデータを格納
    Set dict = CreateObject("Scripting.Dictionary")

    For i = LBound(ipkData, 1) To UBound(ipkData, 1)
        Dim dictKey As String
        dictKey = CStr(ipkData(i, 2)) & "|" & CStr(ipkData(i, 3)) & "|" & CStr(ipkData(i, 4)) & "|" & CStr(ipkData(i, 5))
        dict(dictKey) = True
        wsDebug.Cells(i + 1, 6).Value = ipkData(i, 2)
        wsDebug.Cells(i + 1, 7).Value = ipkData(i, 3)
        wsDebug.Cells(i + 1, 8).Value = ipkData(i, 4)
        wsDebug.Cells(i + 1, 9).Value = ipkData(i, 5)
        wsDebug.Cells(i + 1, 11).Value = dictKey
    Next i

    ' 庶務事務シートのデータを比較し、結果をQ列に記述
    For i = LBound(shomuData, 1) To UBound(shomuData, 1)
        result = "新規口座"
        
        Dim gValue As String
        Dim jValue As String
        If shomuData(i, 5) = 9900 Then
            gValue = Mid(CStr(shomuData(i, 7)), 2, 2) & "8"
            
            If Right(CStr(shomuData(i, 10)), 1) = "1" Then
                jValue = Left(CStr(shomuData(i, 10)), Len(CStr(shomuData(i, 10))) - 1)
            Else
                jValue = CStr(shomuData(i, 10))
            End If
        Else
            gValue = Format(shomuData(i, 7), "000")
            jValue = CStr(shomuData(i, 10))
        End If

        Dim shomuKey As String
        shomuKey = CStr(shomuData(i, 2)) & "|" & CStr(shomuData(i, 5)) & gValue & "|" & CStr(shomuData(i, 9)) & "|" & jValue

        wsDebug.Cells(i + 1, 1).Value = shomuData(i, 2)
        wsDebug.Cells(i + 1, 2).Value = shomuData(i, 5)
        wsDebug.Cells(i + 1, 3).Value = shomuData(i, 7)
        wsDebug.Cells(i + 1, 4).Value = shomuData(i, 9)
        wsDebug.Cells(i + 1, 5).Value = shomuData(i, 10)
        wsDebug.Cells(i + 1, 10).Value = shomuKey

        If dict.Exists(shomuKey) Then
            result = "IPK登録あり"
        End If

        wsShomu.Cells(i + 1, "Q").Value = result
    Next i

    ' 8. 庶務事務シートQ1セルに「確認結果」と記述
    Application.StatusBar = "8. 庶務事務シートQ1セルに「確認結果」と記述"
    wsShomu.Cells(1, "Q").Value = "確認結果"

    ' 9. IPKシートの全列のデータをクリアしてから削除
    Application.StatusBar = "9. IPKシートの全列のデータをクリアしてから削除"
    wsIPK.Cells.ClearContents
    wsIPK.Cells.Delete

    ' 10. 判定結果を踏まえてR列に記述する
    Application.StatusBar = "10. 判定結果を踏まえてR列に記述する"
    Set dict = CreateObject("Scripting.Dictionary")

    For i = 2 To lastRowShomu
        Dim keyStr As String
        keyStr = wsShomu.Cells(i, "A").Value & wsShomu.Cells(i, "B").Value

        If Not dict.Exists(keyStr) Then
            dict.Add keyStr, ""
        End If

        Select Case wsShomu.Cells(i, "D").Value
            Case 1
                accountType = "口座A"
            Case 2
                accountType = "口座B"
            Case 3
                accountType = "口座C"
            Case Else
                accountType = "口座不明"
        End Select

        If dict(keyStr) <> "" Then
            dict(keyStr) = dict(keyStr) & "、"
        End If
        dict(keyStr) = dict(keyStr) & accountType & "：" & wsShomu.Cells(i, "Q").Value
    Next i

    For Each key In dict.Keys
        If InStr(dict(key), "新規口座") > 0 Then
            status = HEADER_KOUZA_ADD
        Else
            status = HEADER_FURIKAE
        End If

        Dim accounts As Variant
        accounts = Split(dict(key), "、")
        SortArray accounts

        Dim sortedAccounts As String
        sortedAccounts = Join(accounts, "、")

        For i = 2 To lastRowShomu
            If wsShomu.Cells(i, "A").Value & wsShomu.Cells(i, "B").Value = key Then
                wsShomu.Cells(i, "R").Value = status & sortedAccounts
            End If
        Next i
    Next key

    ' 11. 庶務事務シートB～Q列のデータをクリアしてから削除
    Application.StatusBar = "11. 庶務事務シートB～Q列のデータをクリアしてから削除"
    wsShomu.Range("B:Q").ClearContents
    wsShomu.Range("B:Q").Delete

    ' 12. 庶務事務シートの重複行を削除する
    Application.StatusBar = "12. 庶務事務シートの重複行を削除する"
    wsShomu.Range("A:B").RemoveDuplicates Columns:=Array(1, 2), Header:=xlYes
    
    ' 13. Debugシートの削除
    Application.StatusBar = "13. Debugシートの削除"
    Application.DisplayAlerts = False
    wsDebug.Delete
    Application.DisplayAlerts = True

    ' 14. RPA用データシートの作成
    Application.StatusBar = "14. RPA用データシートの作成"
    
    ' 庶務事務シートB1セルに「判定結果」と記述
    wsShomu.Cells(1, "B").Value = "判定結果"
    
    ' 「口座追加あり」シートを作成
    Dim wsKouzaAdd As Worksheet
    On Error Resume Next
    Set wsKouzaAdd = ThisWorkbook.Sheets("口座追加あり")
    If wsKouzaAdd Is Nothing Then
        Set wsKouzaAdd = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsKouzaAdd.Name = "口座追加あり"
    End If
    On Error GoTo 0
    wsKouzaAdd.Cells.ClearContents
    wsKouzaAdd.Tab.Color = RGB(255, 255, 0)
    wsKouzaAdd.Cells.Font.Name = "Meiryo"

    ' 「振替額変更」シートを作成
    Dim wsFurikaeChange As Worksheet
    On Error Resume Next
    Set wsFurikaeChange = ThisWorkbook.Sheets("振替額変更")
    If wsFurikaeChange Is Nothing Then
        Set wsFurikaeChange = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsFurikaeChange.Name = "振替額変更"
    End If
    On Error GoTo 0
    wsFurikaeChange.Cells.ClearContents
    wsFurikaeChange.Tab.Color = RGB(255, 255, 0)
    wsFurikaeChange.Cells.Font.Name = "Meiryo"

    ' 「口座追加あり」シートにデータを貼り付け
    Dim kouzaAddRow As Long
    kouzaAddRow = 1
    For i = 2 To lastRowShomu
        If Left(wsShomu.Cells(i, "B").Value, Len(HEADER_KOUZA_ADD)) = HEADER_KOUZA_ADD Then
            wsKouzaAdd.Cells(kouzaAddRow, 1).Value = wsShomu.Cells(i, "A").Value
            wsKouzaAdd.Cells(kouzaAddRow, 2).Value = wsShomu.Cells(i, "B").Value
            kouzaAddRow = kouzaAddRow + 1
        End If
    Next i
    wsKouzaAdd.Columns(1).NumberFormat = "0"

    ' 「振替額変更」シートにデータを貼り付け
    Dim furikaeChangeRow As Long
    furikaeChangeRow = 1
    For i = 2 To lastRowShomu
        If Left(wsShomu.Cells(i, "B").Value, Len(HEADER_FURIKAE)) = HEADER_FURIKAE Then
            wsFurikaeChange.Cells(furikaeChangeRow, 1).Value = wsShomu.Cells(i, "A").Value
            wsFurikaeChange.Cells(furikaeChangeRow, 2).Value = wsShomu.Cells(i, "B").Value
            furikaeChangeRow = furikaeChangeRow + 1
        End If
    Next i
    wsFurikaeChange.Columns(1).NumberFormat = "0"

    ' 15. 庶務事務シートの内容をすべてクリアする
    Application.StatusBar = "15. 庶務事務シートの内容をすべてクリアする"
    wsShomu.Cells.ClearContents

    ' 16. IPKシートのデータを元に戻す
    Application.StatusBar = "16. IPKシートのデータを元に戻す"
    wsIPK.Cells.ClearContents
    wsIPK.Range("A1").Resize(UBound(ipkDataBackup, 1), UBound(ipkDataBackup, 2)).Value = ipkDataBackup

    ' 17. 処理がすべて終了したらメッセージを表示する
    MsgBox "RPA用データが作成されました。", vbInformation

    Application.StatusBar = False
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Exit Sub

ErrorHandler:
    MsgBox "エラーが発生しました。" & vbCrLf & "エラーメッセージ: " & Err.Description, vbCritical
    Application.StatusBar = False
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
End Sub

Sub SortArray(arr As Variant)
    Dim i As Long, j As Long
    Dim temp As String
    For i = LBound(arr) To UBound(arr) - 1
        For j = i + 1 To UBound(arr)
            If arr(i) > arr(j) Then
                temp = arr(i)
                arr(i) = arr(j)
                arr(j) = temp
            End If
        Next j
    Next i
End Sub

    Cells.Select

End Sub
