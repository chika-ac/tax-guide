' ============================================================
' 週次報告書　VBAモジュール（起点セル方式）
' ============================================================
' 【組み込み手順】
'   1. Alt+F11 → 挿入 → 標準モジュール
'   2. このコードを全て貼り付ける
'   3. ボタン①に ReflectDates、ボタン②に PrepareForSubmission を割り当て
' ============================================================

' 転記設定テーブルの位置
Const TABLE_SHEET     As String = "操作パネル"
Const TABLE_ROW_START As Long   = 20   ' データ行開始（ヘッダーの次）
Const COL_KEYWORD     As Long   = 2    ' B：ファイルキーワード
Const COL_SHEET       As Long   = 3    ' C：シート名
Const COL_CELL        As Long   = 4    ' D：起点セル
Const COL_DIR         As Long   = 5    ' E：方向（横/縦）
Const COL_WEEKS       As Long   = 6    ' F：週数（4 or 1）
Const COL_LABEL       As Long   = 7    ' G：転記先ラベル


' ============================================================
' ボタン①　日付反映＋データ取込
' ============================================================
Sub ReflectDates()

    Dim ws_p As Worksheet
    Set ws_p = ThisWorkbook.Worksheets("操作パネル")

    ' 日付チェック
    If IsEmpty(ws_p.Range("C4").Value) Or Not IsDate(ws_p.Range("C4").Value) Then
        MsgBox "C4セルに報告対象週の初日（月曜日）を入力してください。", vbExclamation
        Exit Sub
    End If

    Dim d As Date
    d = CDate(ws_p.Range("C4").Value)

    If Weekday(d, vbMonday) <> 1 Then
        MsgBox "月曜日の日付を入力してください。" & Chr(10) & _
               Format(d, "YYYY/M/D（AAA）"), vbExclamation
        Exit Sub
    End If

    ' 報告対象週を表示
    Dim ws_r As Worksheet
    Set ws_r = ThisWorkbook.Worksheets("報告書")
    With ws_r.Range("C2")
        .Value = "報告対象週：" & Format(d, "M/D（AAA）") & _
                 " 〜 " & Format(d + 4, "M/D（AAA）")
        .Font.Bold = True
        .Font.Italic = False
        .Font.Color = RGB(31, 73, 125)
    End With

    ' 転記実行
    Call ImportByStartCell

    ' 先週比更新
    Call UpdateWeeklyComparison

    MsgBox "完了しました！" & Chr(10) & _
           "報告対象週：" & Format(d, "M/D（AAA）") & _
           " 〜 " & Format(d + 4, "M/D（AAA）"), vbInformation
End Sub


' ============================================================
' 転記設定テーブルを読んで起点セル方式で転記
' ============================================================
Sub ImportByStartCell()

    Dim ws_p As Worksheet
    Set ws_p = ThisWorkbook.Worksheets(TABLE_SHEET)

    Dim last_row As Long
    last_row = ws_p.Cells(ws_p.Rows.Count, COL_KEYWORD).End(xlUp).Row

    ' 開いた外部ファイルをキャッシュ
    Dim cache As Object
    Set cache = CreateObject("Scripting.Dictionary")

    Application.ScreenUpdating = False

    Dim i As Long
    For i = TABLE_ROW_START To last_row

        ' 各列を読む
        Dim kw      As String: kw      = Trim(CStr(ws_p.Cells(i, COL_KEYWORD).Value))
        Dim sh      As String: sh      = Trim(CStr(ws_p.Cells(i, COL_SHEET).Value))
        Dim sc      As String: sc      = Trim(CStr(ws_p.Cells(i, COL_CELL).Value))
        Dim dir     As String: dir     = Trim(CStr(ws_p.Cells(i, COL_DIR).Value))
        Dim wks     As Long:   wks     = CLng(IIf(IsNumeric(ws_p.Cells(i, COL_WEEKS).Value), ws_p.Cells(i, COL_WEEKS).Value, 0))
        Dim lbl     As String: lbl     = Trim(CStr(ws_p.Cells(i, COL_LABEL).Value))

        ' 空行・未入力はスキップ
        If kw = "" Or sc = "" Or wks = 0 Then GoTo NextRow
        If InStr(sc, "起点") > 0 Or InStr(sh, "シート名") > 0 Then GoTo NextRow

        ' 外部ファイルを取得（キャッシュ優先）
        Dim wb_src As Workbook
        If cache.Exists(kw) Then
            Set wb_src = cache(kw)
        Else
            Dim fp As String
            fp = FindFile(kw, ThisWorkbook.Path)
            If fp = "" Then
                MsgBox "「" & kw & "」を含むファイルが見つかりません。" & Chr(10) & _
                       "報告書と同じフォルダに置いてください。（" & i & "行目）", vbExclamation
                GoTo NextRow
            End If
            Set wb_src = Workbooks.Open(fp, ReadOnly:=True)
            cache(kw) = wb_src
        End If

        ' 取込元シート
        Dim ws_src As Worksheet
        On Error Resume Next
        Set ws_src = wb_src.Worksheets(sh)
        On Error GoTo 0
        If ws_src Is Nothing Then
            MsgBox "シート「" & sh & "」が見つかりません。（" & i & "行目）", vbExclamation
            GoTo NextRow
        End If

        ' 転記先行を報告書A列ラベルから検索
        Dim dst_row As Long
        dst_row = FindLabelRow(lbl)
        If dst_row = 0 Then
            ' 特殊ラベルの場合は別処理へ
            Call HandleSpecialLabel(ws_src, sc, dir, wks, lbl)
            GoTo NextRow
        End If

        ' 起点セルから値を読んで報告書に転記
        ' 方向：横＝列方向、縦＝行方向
        ' 週数：4なら20セル（第1〜4週）、1なら5セル（第4週のみ）
        Dim total_cells As Long: total_cells = wks * 5

        ' 転記先の開始列：週数4なら第1週月曜（C列=3）、1なら第4週月曜（R列=18）
        Dim dst_col_start As Long
        dst_col_start = IIf(wks = 4, 3, 18)

        Dim j As Long
        For j = 0 To total_cells - 1
            Dim src_val As Variant
            If dir = "横" Then
                src_val = ws_src.Range(sc).Offset(0, j).Value
            ElseIf dir = "縦" Then
                src_val = ws_src.Range(sc).Offset(j, 0).Value
            Else
                src_val = ws_src.Range(sc).Offset(0, j).Value  ' デフォルト横
            End If

            Dim ws_r As Worksheet
            Set ws_r = ThisWorkbook.Worksheets("報告書")
            ws_r.Cells(dst_row, dst_col_start + j).Value = src_val
        Next j

NextRow:
        Set ws_src = Nothing
    Next i

    ' 外部ファイルを全て閉じる
    Dim k As Variant
    For Each k In cache.Keys
        cache(k).Close SaveChanges:=False
    Next k

    Application.ScreenUpdating = True
End Sub


' ============================================================
' 特殊ラベル（時間帯別・カテゴリ週比較・区局別）の転記
' ============================================================
Sub HandleSpecialLabel(ws_src As Worksheet, sc As String, _
                       dir As String, wks As Long, lbl As String)

    Dim ws_d As Worksheet

    Select Case True

    Case InStr(lbl, "時間帯別") > 0
        ' 詳細データシートの時間帯別エリアへ（10行×5列）
        Set ws_d = ThisWorkbook.Worksheets("詳細データ")
        Dim t_row As Long
        t_row = FindLabelRowInSheet(ws_d, "⑥ 時間帯別") + 3  ' データ開始行
        Dim r As Long, c As Long
        For r = 0 To 9      ' 10時間帯
            For c = 0 To 4  ' 月〜金
                Dim v As Variant
                If dir = "横" Then
                    v = ws_src.Range(sc).Offset(r, c).Value
                Else
                    v = ws_src.Range(sc).Offset(r * 5 + c, 0).Value
                End If
                ws_d.Cells(t_row + r, 2 + c).Value = v  ' B〜F列
            Next c
        Next r

    Case InStr(lbl, "カテゴリ別_今週") > 0
        ' 詳細データシートのカテゴリ週比較 B列へ（12行）
        Set ws_d = ThisWorkbook.Worksheets("詳細データ")
        Dim c5_row As Long
        c5_row = FindLabelRowInSheet(ws_d, "⑤ カテゴリ別") + 3
        Dim ci As Long
        For ci = 0 To 11
            If dir = "横" Then
                ws_d.Cells(c5_row + ci, 2).Value = ws_src.Range(sc).Offset(0, ci).Value
            Else
                ws_d.Cells(c5_row + ci, 2).Value = ws_src.Range(sc).Offset(ci, 0).Value
            End If
        Next ci

    Case InStr(lbl, "カテゴリ別_先週") > 0
        Set ws_d = ThisWorkbook.Worksheets("詳細データ")
        Dim c5_row2 As Long
        c5_row2 = FindLabelRowInSheet(ws_d, "⑤ カテゴリ別") + 3
        Dim ci2 As Long
        For ci2 = 0 To 11
            If dir = "横" Then
                ws_d.Cells(c5_row2 + ci2, 3).Value = ws_src.Range(sc).Offset(0, ci2).Value
            Else
                ws_d.Cells(c5_row2 + ci2, 3).Value = ws_src.Range(sc).Offset(ci2, 0).Value
            End If
        Next ci2

    Case InStr(lbl, "区局別") > 0
        ' 詳細データシートの区局別エリア（20行×5列）
        Set ws_d = ThisWorkbook.Worksheets("詳細データ")
        Dim w_row As Long
        w_row = FindLabelRowInSheet(ws_d, "⑦ 問い合わせ") + 3
        Dim wr As Long, wc As Long
        For wr = 0 To 19
            For wc = 0 To 4
                If dir = "横" Then
                    ws_d.Cells(w_row + wr, 3 + wc).Value = _
                        ws_src.Range(sc).Offset(wr, wc).Value
                Else
                    ws_d.Cells(w_row + wr, 3 + wc).Value = _
                        ws_src.Range(sc).Offset(wr * 5 + wc, 0).Value
                End If
            Next wc
        Next wr

    End Select
End Sub


' ============================================================
' 同フォルダからキーワードを含むExcelファイルを検索
' ============================================================
Function FindFile(keyword As String, folder As String) As String
    Dim f As String
    For Each ext In Array("*.xlsx", "*.xlsm")
        f = Dir(folder & "\" & ext)
        Do While f <> ""
            If InStr(f, keyword) > 0 And f <> ThisWorkbook.Name Then
                FindFile = folder & "\" & f: Exit Function
            End If
            f = Dir()
        Loop
    Next
    FindFile = ""
End Function


' ============================================================
' 報告書シートのA列からラベルを検索して行番号を返す
' ============================================================
Function FindLabelRow(label As String) As Long
    FindLabelRow = FindLabelRowInSheet(ThisWorkbook.Worksheets("報告書"), label)
End Function

Function FindLabelRowInSheet(ws As Worksheet, label As String) As Long
    Dim r As Long
    For r = 1 To 300
        If InStr(CStr(ws.Cells(r, 1).Value), label) > 0 Then
            FindLabelRowInSheet = r: Exit Function
        End If
    Next r
    FindLabelRowInSheet = 0
End Function


' ============================================================
' 先週比をKPIサマリに反映
' ============================================================
Sub UpdateWeeklyComparison()

    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("報告書")

    ' 各項目の行番号をラベルから取得
    Dim r_zentai As Long: r_zentai = FindLabelRow("⑤全体対応件数")
    Dim r_nyuden As Long: r_nyuden = FindLabelRow("①入電件数")
    Dim r_hassin As Long: r_hassin = FindLabelRow("④発信件数")
    Dim r_houki  As Long: r_houki  = FindLabelRow("　②放棄呼数")
    Dim r_stage1 As Long: r_stage1 = FindLabelRow("第一段階")
    Dim r_stage2 As Long: r_stage2 = FindLabelRow("第二段階")

    ' KPI1段目の先週比セル（先週比ラベルが入っている行 = KPI値行 + 1）
    Dim r_kpi1 As Long: r_kpi1 = FindLabelRow("全体対応件数") + 2

    ' 第4週(R〜V=18〜22列)と第3週(M〜Q=13〜17列)を比較
    Dim items As Variant
    items = Array( _
        Array(r_zentai, "A"), _
        Array(r_nyuden, "D"), _
        Array(r_hassin, "G"), _
        Array(r_houki,  "J")  _
    )

    Dim it As Variant
    For Each it In items
        Dim row_num As Long: row_num = it(0)
        Dim kpi_col As String: kpi_col = it(1)
        If row_num > 0 Then
            Dim n As Long: n = WorksheetFunction.Sum(ws.Range("R" & row_num & ":V" & row_num))
            Dim p As Long: p = WorksheetFunction.Sum(ws.Range("M" & row_num & ":Q" & row_num))
            ws.Range(kpi_col & r_kpi1).Value = FormatDiff(n - p) & "件"
        End If
    Next it

    ' 呼損率
    If r_nyuden > 0 And r_houki > 0 Then
        Dim nn As Long: nn = WorksheetFunction.Sum(ws.Range("R" & r_nyuden & ":V" & r_nyuden))
        Dim np As Long: np = WorksheetFunction.Sum(ws.Range("M" & r_nyuden & ":Q" & r_nyuden))
        Dim hn As Long: hn = WorksheetFunction.Sum(ws.Range("R" & r_houki & ":V" & r_houki))
        Dim hp As Long: hp = WorksheetFunction.Sum(ws.Range("M" & r_houki & ":Q" & r_houki))
        If nn > 0 And np > 0 Then
            Dim rd As Double: rd = (hn / nn - hp / np) * 100
            ws.Range("M" & r_kpi1).Value = FormatDiff(Format(rd, "0.0")) & "%"
        End If
    End If

    ' 第一・第二段階
    Dim r_kpi2 As Long: r_kpi2 = FindLabelRow("稼働日平均受電件数") + 2
    For Each it In Array(Array(r_stage1, "E"), Array(r_stage2, "I"))
        If it(0) > 0 Then
            Dim sn As Long: sn = WorksheetFunction.Sum(ws.Range("R" & it(0) & ":V" & it(0)))
            Dim sp As Long: sp = WorksheetFunction.Sum(ws.Range("M" & it(0) & ":Q" & it(0)))
            ws.Range(it(1) & r_kpi2).Value = FormatDiff(sn - sp) & "件"
        End If
    Next it
End Sub


' ============================================================
' 差分を「先週比：▲N」形式に変換
' ============================================================
Function FormatDiff(diff As Variant) As String
    Dim n As Double: n = CDbl(diff)
    If n > 0 Then
        FormatDiff = "先週比：▲" & diff
    ElseIf n < 0 Then
        FormatDiff = "先週比：▼" & Abs(diff)
    Else
        FormatDiff = "先週比：±0"
    End If
End Function


' ============================================================
' ボタン②　提出用に整える（操作パネル・作業データ削除）
' ============================================================
Sub PrepareForSubmission()

    If MsgBox("操作パネルと作業データシートを削除して提出用ファイルにします。" & Chr(10) & _
              "この操作は元に戻せません。よろしいですか？", _
              vbYesNo + vbExclamation, "提出用に整える") = vbNo Then Exit Sub

    Application.DisplayAlerts = False
    Dim nm As Variant
    For Each nm In Array("操作パネル", "作業データ")
        On Error Resume Next
        ThisWorkbook.Worksheets(nm).Delete
        On Error GoTo 0
    Next nm
    Application.DisplayAlerts = True

    ThisWorkbook.Worksheets("報告書").Activate
    ThisWorkbook.Save
    MsgBox "完了しました。", vbInformation
End Sub
