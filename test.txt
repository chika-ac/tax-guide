Sub ExportFAQ()
    Dim ws As Worksheet
    Dim fPath As Variant
    Dim fNum As Integer
    Dim lastRow As Long
    Dim i As Long
    Dim j As Integer
    Dim line As String
    Dim val As String

    Set ws = ThisWorkbook.Sheets("FAQ")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lastRow < 2 Then
        MsgBox "Error: No data in FAQ sheet."
        Exit Sub
    End If

    fPath = Application.GetSaveAsFilename( _
        InitialFileName:="FAQ_export_" & Format(Now, "YYYYMMDD") & ".csv", _
        FileFilter:="CSV," & Chr(42) & ".csv")
    If VarType(fPath) = vbBoolean Then Exit Sub

    fNum = FreeFile
    Open fPath For Output As #fNum

    For i = 1 To lastRow
        line = ""
        For j = 1 To 5
            val = CStr(ws.Cells(i, j).Value)
            val = Replace(val, Chr(34), Chr(34) & Chr(34))
            If InStr(val, Chr(44)) > 0 Or InStr(val, Chr(10)) > 0 Or InStr(val, Chr(34)) > 0 Then
                val = Chr(34) & val & Chr(34)
            End If
            If j = 1 Then
                line = val
            Else
                line = line & Chr(44) & val
            End If
        Next j
        Print #fNum, line
    Next i

    Close #fNum

    MsgBox "FAQ_export wo hozon shimashita."
End Sub
