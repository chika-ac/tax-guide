あSub ExportFinalFAQ()
    Dim wsFAQ As Worksheet
    Dim wsDup As Worksheet
    Dim wsFinal As Worksheet
    Dim lastRowFAQ As Long
    Dim lastRowDup As Long
    Dim i As Long
    Dim d As Long
    Dim outRow As Long
    Dim action As String
    Dim newQ As String
    Dim newA As String
    Dim newKw As String
    Dim newShin As String
    Dim rowA As Long
    Dim rowB As Long

    Set wsFAQ = ThisWorkbook.Sheets("FAQ")

    ' 重複チェックシート確認
    Dim sheetExists As Boolean
    sheetExists = False
    Dim s As Worksheet
    For Each s In ThisWorkbook.Sheets
        If s.Name = "重複チェック" Then
            sheetExists = True
            Exit For
        End If
    Next s

    If Not sheetExists Then
        MsgBox "Error: sheet not found. Please run DetectDuplicates first."
        Exit Sub
    End If

    Set wsDup = ThisWorkbook.Sheets("重複チェック")

    ' FinalFAQシートを作成または初期化
    Dim finalExists As Boolean
    finalExists = False
    For Each s In ThisWorkbook.Sheets
        If s.Name = "FinalFAQ" Then
            finalExists = True
            Exit For
        End If
    Next s

    If finalExists Then
        Set wsFinal = ThisWorkbook.Sheets("FinalFAQ")
        wsFinal.Cells.ClearContents
        wsFinal.Cells.Interior.ColorIndex = xlNone
    Else
        Set wsFinal = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        wsFinal.Name = "FinalFAQ"
    End If

    ' ヘッダーコピー
    wsFAQ.Rows(1).Copy wsFinal.Rows(1)

    lastRowFAQ = wsFAQ.Cells(wsFAQ.Rows.Count, 1).End(xlUp).Row
    lastRowDup = wsDup.Cells(wsDup.Rows.Count, 1).End(xlUp).Row

    Application.ScreenUpdating = False

    ' ----------------------------------------------------------------
    ' STEP1：残すフラグ・削除フラグを別々に管理
    ' ----------------------------------------------------------------
    Dim keepFlag() As Boolean
    Dim deleteFlag() As Boolean
    ReDim keepFlag(lastRowFAQ)
    ReDim deleteFlag(lastRowFAQ)

    Dim newData() As String
    ReDim newData(lastRowDup, 4)

    For d = 2 To lastRowDup
        rowA = CLng(wsDup.Cells(d, 2).Value)
        rowB = CLng(wsDup.Cells(d, 7).Value)
        action = Trim(wsDup.Cells(d, 12).Value)
        If action = "" Then action = "Ryoho_Nokosu"

        Select Case action
            Case "A_Nokosu"
                keepFlag(rowA) = True
                deleteFlag(rowB) = True

            Case "B_Nokosu"
                keepFlag(rowB) = True
                deleteFlag(rowA) = True

            Case "Ryoho_Nokosu"
                keepFlag(rowA) = True
                keepFlag(rowB) = True

            Case "Ryoho_Sakujo_Shinkiseisei"
                deleteFlag(rowA) = True
                deleteFlag(rowB) = True
                newData(d, 0) = wsFAQ.Cells(rowA, 1).Value
                newData(d, 1) = Trim(wsDup.Cells(d, 13).Value)
                newData(d, 2) = Trim(wsDup.Cells(d, 14).Value)
                newData(d, 3) = Trim(wsDup.Cells(d, 15).Value)
                newData(d, 4) = Trim(wsDup.Cells(d, 16).Value)

            Case "Ryoho_Sakujo"
                deleteFlag(rowA) = True
                deleteFlag(rowB) = True
        End Select
    Next d

    ' ----------------------------------------------------------------
    ' STEP2：FinalFAQシートに出力
    ' ----------------------------------------------------------------
    outRow = 2

    For i = 2 To lastRowFAQ
        ' 残すフラグあり、またはどちらのフラグもない（重複に未登場）→ 出力
        ' 削除フラグあり かつ 残すフラグなし → 出力しない
        If keepFlag(i) Or Not deleteFlag(i) Then
            wsFAQ.Rows(i).Copy wsFinal.Rows(outRow)
            outRow = outRow + 1
        End If
    Next i

    ' 新規作成データを出力
    For d = 2 To lastRowDup
        If newData(d, 1) <> "" Then
            wsFinal.Cells(outRow, 1).Value = newData(d, 0)
            wsFinal.Cells(outRow, 2).Value = newData(d, 1)
            wsFinal.Cells(outRow, 3).Value = newData(d, 2)
            wsFinal.Cells(outRow, 4).Value = newData(d, 3)
            wsFinal.Cells(outRow, 5).Value = newData(d, 4)
            wsFinal.Cells(outRow, 6).Value = Format(Now, "YYYY/MM/DD")
            wsFinal.Rows(outRow).Font.Name = "Arial"
            wsFinal.Rows(outRow).Font.Size = 9
            wsFinal.Cells(outRow, 3).WrapText = True
            wsFinal.Rows(outRow).RowHeight = 60
            outRow = outRow + 1
        End If
    Next d

    ' 列幅調整
    wsFinal.Columns(1).ColumnWidth = 18
    wsFinal.Columns(2).ColumnWidth = 35
    wsFinal.Columns(3).ColumnWidth = 60
    wsFinal.Columns(4).ColumnWidth = 25
    wsFinal.Columns(5).ColumnWidth = 10
    wsFinal.Columns(6).ColumnWidth = 12

    Application.ScreenUpdating = True

    MsgBox outRow - 2 & " ken no FAQ wo FinalFAQ sheet ni shutsuryoku shimashita."
End Sub
