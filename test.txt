' ============================================================
' 週次報告書　VBAモジュール
' ファイル：02_01ヘルプデスク運用に関する週次報告書_work.xlsx
' ============================================================
' 【組み込み手順】
'   1. Alt+F11 でVBAエディタを開く
'   2. 挿入 → 標準モジュール
'   3. このコードを全て貼り付ける
'   4. 操作パネルのボタン①に ReflectDates を割り当て
'   5. 操作パネルのボタン②に PrepareForSubmission を割り当て
' ============================================================

' ------------------------------------------------------------
' 定数：転記設定テーブルの場所（操作パネルシート）
' ------------------------------------------------------------
Const TABLE_SHEET     As String = "操作パネル"
Const TABLE_ROW_START As Long = 19  ' データ行の開始行（ヘッダーの次）
Const COL_SRC_KEYWORD As Long = 2   ' B列：取込元ファイルキーワード
Const COL_SRC_SHEET   As Long = 3   ' C列：取込元シート名
Const COL_SRC_CELL    As Long = 4   ' D列：取込元セル
Const COL_DST_SHEET   As Long = 5   ' E列：転記先シート名
Const COL_DST_CELL    As Long = 6   ' F列：転記先セル


' ============================================================
' ① データ取込・報告書に反映
' ============================================================
Sub ReflectDates()

    Dim ws_panel As Worksheet
    Dim start_date As Date

    Set ws_panel = ThisWorkbook.Worksheets("操作パネル")

    ' --- 日付チェック ---
    If IsEmpty(ws_panel.Range("C4").Value) Or Not IsDate(ws_panel.Range("C4").Value) Then
        MsgBox "操作パネルのC4セルに報告対象週の初日（月曜日）を入力してください。", vbExclamation
        Exit Sub
    End If

    start_date = CDate(ws_panel.Range("C4").Value)

    If Weekday(start_date, vbMonday) <> 1 Then
        MsgBox "月曜日の日付を入力してください。" & Chr(10) & _
               "入力値：" & Format(start_date, "YYYY/M/D") & _
               "（" & Format(start_date, "AAA") & "）", vbExclamation
        Exit Sub
    End If

    ' --- 報告対象週を報告書シートに表示 ---
    Dim ws_report As Worksheet
    Set ws_report = ThisWorkbook.Worksheets("報告書")
    ws_report.Range("C2").Value = _
        "報告対象週：" & Format(start_date, "M/D（AAA）") & _
        " 〜 " & Format(start_date + 4, "M/D（AAA）")
    ws_report.Range("C2").Font.Bold = True
    ws_report.Range("C2").Font.Italic = False
    ws_report.Range("C2").Font.Color = RGB(31, 73, 125)

    ' --- 転記設定テーブルを読んでデータ取込 ---
    Call ImportFromTable

    ' --- 先週比を更新 ---
    Call UpdateWeeklyComparison

    MsgBox "完了しました！" & Chr(10) & Chr(10) & _
           "報告対象週：" & Format(start_date, "M/D（AAA）") & _
           " 〜 " & Format(start_date + 4, "M/D（AAA）"), vbInformation

End Sub


' ============================================================
' 転記設定テーブルを上から順に読んで転記するメイン処理
' ============================================================
Sub ImportFromTable()

    Dim ws_panel As Worksheet
    Set ws_panel = ThisWorkbook.Worksheets(TABLE_SHEET)

    ' テーブルの最終行を取得
    Dim last_row As Long
    last_row = ws_panel.Cells(ws_panel.Rows.Count, COL_SRC_KEYWORD).End(xlUp).Row

    ' 開いた外部ファイルをキャッシュ（同じファイルを何度も開かないため）
    Dim opened_files As Object
    Set opened_files = CreateObject("Scripting.Dictionary")

    Dim i As Long
    For i = TABLE_ROW_START To last_row

        Dim kw        As String
        Dim src_sheet As String
        Dim src_cell  As String
        Dim dst_sheet As String
        Dim dst_cell  As String

        kw        = Trim(CStr(ws_panel.Cells(i, COL_SRC_KEYWORD).Value))
        src_sheet = Trim(CStr(ws_panel.Cells(i, COL_SRC_SHEET).Value))
        src_cell  = Trim(CStr(ws_panel.Cells(i, COL_SRC_CELL).Value))
        dst_sheet = Trim(CStr(ws_panel.Cells(i, COL_DST_SHEET).Value))
        dst_cell  = Trim(CStr(ws_panel.Cells(i, COL_DST_CELL).Value))

        ' 空行・未入力プレースホルダーはスキップ
        If kw = "" Or src_cell = "" Then GoTo NextRow
        If InStr(src_cell, "入力") > 0 Or InStr(src_sheet, "入力") > 0 Then GoTo NextRow

        ' 転記先シートを取得
        Dim ws_dst As Worksheet
        On Error Resume Next
        Set ws_dst = ThisWorkbook.Worksheets(dst_sheet)
        On Error GoTo 0
        If ws_dst Is Nothing Then
            MsgBox "転記先シート「" & dst_sheet & "」が見つかりません。（" & i & "行目）", vbExclamation
            GoTo NextRow
        End If

        ' 外部ファイルをキャッシュまたは新規オープン
        Dim wb_src As Workbook
        If opened_files.Exists(kw) Then
            Set wb_src = opened_files(kw)
        Else
            Dim found_path As String
            found_path = FindFileByKeyword(kw, ThisWorkbook.Path)

            If found_path = "" Then
                MsgBox "「" & kw & "」を含むExcelファイルが見つかりません。" & Chr(10) & _
                       "報告書と同じフォルダに置いてください。（" & i & "行目）", vbExclamation
                GoTo NextRow
            End If

            Application.ScreenUpdating = False
            Set wb_src = Workbooks.Open(found_path, ReadOnly:=True)
            opened_files(kw) = wb_src
        End If

        ' 取込元シートを取得
        Dim ws_src As Worksheet
        On Error Resume Next
        Set ws_src = wb_src.Worksheets(src_sheet)
        On Error GoTo 0
        If ws_src Is Nothing Then
            MsgBox "取込元シート「" & src_sheet & "」が見つかりません。" & Chr(10) & _
                   "ファイル：" & wb_src.Name & "（" & i & "行目）", vbExclamation
            GoTo NextRow
        End If

        ' 値を転記
        ws_dst.Range(dst_cell).Value = ws_src.Range(src_cell).Value

NextRow:
        Set ws_src = Nothing
        Set ws_dst = Nothing
    Next i

    ' 開いた外部ファイルを全て閉じる
    Dim key As Variant
    For Each key In opened_files.Keys
        opened_files(key).Close SaveChanges:=False
    Next key

    Application.ScreenUpdating = True

End Sub


' ============================================================
' 同フォルダからキーワードを含むExcelファイルを検索
' ============================================================
Function FindFileByKeyword(keyword As String, folder_path As String) As String

    Dim file_name As String

    ' .xlsx を検索
    file_name = Dir(folder_path & "\*.xlsx")
    Do While file_name <> ""
        If InStr(file_name, keyword) > 0 And file_name <> ThisWorkbook.Name Then
            FindFileByKeyword = folder_path & "\" & file_name
            Exit Function
        End If
        file_name = Dir()
    Loop

    ' .xlsm も検索
    file_name = Dir(folder_path & "\*.xlsm")
    Do While file_name <> ""
        If InStr(file_name, keyword) > 0 And file_name <> ThisWorkbook.Name Then
            FindFileByKeyword = folder_path & "\" & file_name
            Exit Function
        End If
        file_name = Dir()
    Loop

    FindFileByKeyword = ""

End Function


' ============================================================
' 先週比をKPIサマリに反映
' ============================================================
Sub UpdateWeeklyComparison()

    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("報告書")

    ' 各行番号をラベルから動的に取得
    Dim r_zentai As Long: r_zentai = FindRowByLabel(ws, "⑤全体対応件数")
    Dim r_nyuden As Long: r_nyuden = FindRowByLabel(ws, "①入電件数")
    Dim r_hassin As Long: r_hassin = FindRowByLabel(ws, "④発信件数")
    Dim r_houki  As Long: r_houki  = FindRowByLabel(ws, "　②放棄呼数")
    Dim r_stage1 As Long: r_stage1 = FindRowByLabel(ws, "第一段階")
    Dim r_stage2 As Long: r_stage2 = FindRowByLabel(ws, "第二段階")

    ' 第4週(R〜V列)と第3週(M〜Q列)の合計を比較
    Dim updates(5, 1) As Variant  ' (行番号, 差分)
    Dim kpi1_cells As Variant
    kpi1_cells = Array("A9", "D9", "G9", "J9")  ' 全体/入電/発信/放棄呼の先週比セル
    Dim kpi2_cells As Variant
    kpi2_cells = Array("E14", "I14")             ' 第一段階/第二段階の先週比セル

    ' KPI1段目
    Dim rows1 As Variant
    rows1 = Array(r_zentai, r_nyuden, r_hassin, r_houki)
    Dim j As Integer
    For j = 0 To 3
        If rows1(j) > 0 Then
            Dim now_val  As Long: now_val  = WorksheetFunction.Sum(ws.Range("R" & rows1(j) & ":V" & rows1(j)))
            Dim prev_val As Long: prev_val = WorksheetFunction.Sum(ws.Range("M" & rows1(j) & ":Q" & rows1(j)))
            ws.Range(kpi1_cells(j)).Value = FormatDiff(now_val - prev_val) & "件"
        End If
    Next j

    ' 呼損率の先週比
    If r_nyuden > 0 And r_houki > 0 Then
        Dim nyuden_now  As Long: nyuden_now  = WorksheetFunction.Sum(ws.Range("R" & r_nyuden & ":V" & r_nyuden))
        Dim nyuden_prev As Long: nyuden_prev = WorksheetFunction.Sum(ws.Range("M" & r_nyuden & ":Q" & r_nyuden))
        Dim houki_now   As Long: houki_now   = WorksheetFunction.Sum(ws.Range("R" & r_houki & ":V" & r_houki))
        Dim houki_prev  As Long: houki_prev  = WorksheetFunction.Sum(ws.Range("M" & r_houki & ":Q" & r_houki))
        Dim rate_diff As Double
        If nyuden_now > 0 And nyuden_prev > 0 Then
            rate_diff = (houki_now / nyuden_now - houki_prev / nyuden_prev) * 100
            ws.Range("M9").Value = FormatDiff(Format(rate_diff, "0.0")) & "%"
        End If
    End If

    ' KPI2段目（第一・第二段階）
    Dim rows2 As Variant
    rows2 = Array(r_stage1, r_stage2)
    For j = 0 To 1
        If rows2(j) > 0 Then
            Dim s_now  As Long: s_now  = WorksheetFunction.Sum(ws.Range("R" & rows2(j) & ":V" & rows2(j)))
            Dim s_prev As Long: s_prev = WorksheetFunction.Sum(ws.Range("M" & rows2(j) & ":Q" & rows2(j)))
            ws.Range(kpi2_cells(j)).Value = FormatDiff(s_now - s_prev) & "件"
        End If
    Next j

End Sub


' ============================================================
' 差分を「先週比：▲N」「先週比：▼N」形式にフォーマット
' ============================================================
Function FormatDiff(diff As Variant) As String
    Dim n As Double
    n = CDbl(diff)
    If n > 0 Then
        FormatDiff = "先週比：▲" & diff
    ElseIf n < 0 Then
        FormatDiff = "先週比：▼" & Abs(diff)
    Else
        FormatDiff = "先週比：±0"
    End If
End Function


' ============================================================
' 報告書シートのA列からラベルを検索して行番号を返す
' ============================================================
Function FindRowByLabel(ws As Worksheet, label As String) As Long
    Dim r As Long
    For r = 1 To 200
        If InStr(CStr(ws.Cells(r, 1).Value), label) > 0 Then
            FindRowByLabel = r
            Exit Function
        End If
    Next r
    FindRowByLabel = 0
End Function


' ============================================================
' ② 提出用に整える（操作パネル・作業データ削除）
' ============================================================
Sub PrepareForSubmission()

    Dim answer As Integer
    answer = MsgBox( _
        "操作パネルと作業データシートを削除して提出用ファイルにします。" & Chr(10) & _
        "この操作は元に戻せません。よろしいですか？", _
        vbYesNo + vbExclamation, "提出用に整える")

    If answer = vbNo Then Exit Sub

    Application.DisplayAlerts = False

    Dim nm As Variant
    For Each nm In Array("操作パネル", "作業データ")
        On Error Resume Next
        ThisWorkbook.Worksheets(nm).Delete
        On Error GoTo 0
    Next nm

    Application.DisplayAlerts = True

    ThisWorkbook.Worksheets("報告書").Activate
    ThisWorkbook.Save

    MsgBox "完了しました。提出用ファイルの準備ができました。", vbInformation

End Sub
