For i = 1 To ACCESS_COUNT
    ' シートが存在する場合のみクリア
    Dim wsExists As Boolean
    wsExists = False
    Dim wsCheck As Worksheet
    For Each wsCheck In ThisWorkbook.Sheets
        If wsCheck.Name = Accessシート名(i) Then
            wsExists = True
            Exit For
        End If
    Next wsCheck

    If wsExists Then
        Set ws = ThisWorkbook.Sheets(Accessシート名(i))
        Call シートクリア(ws, 2)
    End If
Next i
