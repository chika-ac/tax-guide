' ============================================================
' 週次報告書　VBAモジュール（行取込＋範囲コピー対応）
' ============================================================
' 【組み込み手順】
'   1. Alt+F11 → 挿入 → 標準モジュール
'   2. このコードを全て貼り付ける
'   3. ボタン①に ReflectDates、ボタン②に PrepareForSubmission を割り当て
' ============================================================

' 転記設定テーブルの位置（操作パネルシート）
Const TABLE_SHEET     As String = "操作パネル"
Const TABLE_ROW_START As Long   = 20   ' データ行開始
Const COL_KEYWORD     As Long   = 2    ' B：ファイルキーワード
Const COL_SHEET       As Long   = 3    ' C：シート名
Const COL_CELL        As Long   = 4    ' D：起点セル（左上）
Const COL_DIR         As Long   = 5    ' E：方向（横/縦/－）
Const COL_WEEKS       As Long   = 6    ' F：週数（4/1/－）
Const COL_LABEL       As Long   = 7    ' G：転記先（ラベル or セル番地）
Const COL_MODE        As Long   = 8    ' H：転記モード（行取込/範囲コピー）


' ============================================================
' ボタン①　日付反映＋データ取込
' ============================================================
Sub ReflectDates()

    Dim ws_p As Worksheet
    Set ws_p = ThisWorkbook.Worksheets("操作パネル")

    If IsEmpty(ws_p.Range("C4").Value) Or Not IsDate(ws_p.Range("C4").Value) Then
        MsgBox "C4セルに報告対象週の初日（月曜日）を入力してください。", vbExclamation
        Exit Sub
    End If

    Dim d As Date
    d = CDate(ws_p.Range("C4").Value)

    If Weekday(d, vbMonday) <> 1 Then
        MsgBox "月曜日の日付を入力してください。" & Chr(10) & _
               Format(d, "YYYY/M/D（AAA）"), vbExclamation
        Exit Sub
    End If

    Dim ws_r As Worksheet
    Set ws_r = ThisWorkbook.Worksheets("報告書")
    With ws_r.Range("C2")
        .Value = "報告対象週：" & Format(d, "M/D（AAA）") & _
                 " 〜 " & Format(d + 4, "M/D（AAA）")
        .Font.Bold = True
        .Font.Italic = False
        .Font.Color = RGB(31, 73, 125)
    End With

    Call ImportByTable
    Call UpdateWeeklyComparison

    MsgBox "完了しました！" & Chr(10) & _
           "報告対象週：" & Format(d, "M/D（AAA）") & _
           " 〜 " & Format(d + 4, "M/D（AAA）"), vbInformation
End Sub


' ============================================================
' 設定テーブルを読んで転記（行取込／範囲コピー 両対応）
' ============================================================
Sub ImportByTable()

    Dim ws_p As Worksheet
    Set ws_p = ThisWorkbook.Worksheets(TABLE_SHEET)

    Dim last_row As Long
    last_row = ws_p.Cells(ws_p.Rows.Count, COL_KEYWORD).End(xlUp).Row

    Dim cache As Object
    Set cache = CreateObject("Scripting.Dictionary")

    Application.ScreenUpdating = False

    Dim i As Long
    For i = TABLE_ROW_START To last_row

        Dim kw   As String: kw   = Trim(CStr(ws_p.Cells(i, COL_KEYWORD).Value))
        Dim sh   As String: sh   = Trim(CStr(ws_p.Cells(i, COL_SHEET).Value))
        Dim sc   As String: sc   = Trim(CStr(ws_p.Cells(i, COL_CELL).Value))
        Dim dir  As String: dir  = Trim(CStr(ws_p.Cells(i, COL_DIR).Value))
        Dim wks  As String: wks  = Trim(CStr(ws_p.Cells(i, COL_WEEKS).Value))
        Dim lbl  As String: lbl  = Trim(CStr(ws_p.Cells(i, COL_LABEL).Value))
        Dim mode As String: mode = Trim(CStr(ws_p.Cells(i, COL_MODE).Value))

        ' 空行・未入力はスキップ
        If kw = "" Or sc = "" Or mode = "" Then GoTo NextRow
        If InStr(sc, "起点") > 0 Or InStr(sh, "シート名") > 0 Then GoTo NextRow

        ' 外部ファイルをキャッシュ
        Dim wb_src As Workbook
        If cache.Exists(kw) Then
            Set wb_src = cache(kw)
        Else
            Dim fp As String
            fp = FindFile(kw, ThisWorkbook.Path)
            If fp = "" Then
                MsgBox "「" & kw & "」を含むファイルが見つかりません。" & Chr(10) & _
                       "報告書と同じフォルダに置いてください。（" & i & "行目）", vbExclamation
                GoTo NextRow
            End If
            Set wb_src = Workbooks.Open(fp, ReadOnly:=True)
            cache.Add kw, wb_src
        End If

        ' 取込元シート
        Dim ws_src As Worksheet
        On Error Resume Next
        Set ws_src = wb_src.Worksheets(sh)
        On Error GoTo 0
        If ws_src Is Nothing Then
            MsgBox "シート「" & sh & "」が見つかりません。（" & i & "行目）", vbExclamation
            GoTo NextRow
        End If

        ' ===== モードで分岐 =====
        Select Case mode

        ' --------------------------------------------------
        ' 行取込：起点セルから方向×週数分を1行転記
        ' --------------------------------------------------
        Case "行取込"
            Dim total As Long
            total = IIf(IsNumeric(wks), CLng(wks) * 5, 0)
            If total = 0 Then GoTo NextRow

            ' 転記先行をラベルから検索
            Dim dst_row As Long
            dst_row = FindLabelRow(lbl)

            If dst_row > 0 Then
                ' 報告書への行転記
                Dim dst_col_start As Long
                dst_col_start = IIf(CLng(wks) = 4, 3, 18)  ' 4週:C列、1週:R列
                Dim j As Long
                For j = 0 To total - 1
                    Dim v As Variant
                    v = IIf(dir = "縦", _
                            ws_src.Range(sc).Offset(j, 0).Value, _
                            ws_src.Range(sc).Offset(0, j).Value)
                    ThisWorkbook.Worksheets("報告書").Cells(dst_row, dst_col_start + j).Value = v
                Next j
            Else
                ' 特殊ラベル（詳細データシート向け）
                Call HandleSpecialLabel(ws_src, sc, dir, CLng(IIf(IsNumeric(wks), wks, 1)), lbl)
            End If

        ' --------------------------------------------------
        ' 範囲コピー：起点セルを左上として転記先セルに丸ごとコピー
        ' --------------------------------------------------
        Case "範囲コピー"
            ' lbl に転記先セル番地が入っている（例："C38"）
            If lbl = "" Then GoTo NextRow

            ' 転記先シートを特定（転記先セルの前にシート名があれば使う 例："報告書!C38"）
            Dim dst_ws_name As String
            Dim dst_cell_addr As String
            If InStr(lbl, "!") > 0 Then
                dst_ws_name  = Split(lbl, "!")(0)
                dst_cell_addr = Split(lbl, "!")(1)
            Else
                dst_ws_name  = "報告書"   ' デフォルトは報告書シート
                dst_cell_addr = lbl
            End If

            Dim ws_dst As Worksheet
            On Error Resume Next
            Set ws_dst = ThisWorkbook.Worksheets(dst_ws_name)
            On Error GoTo 0
            If ws_dst Is Nothing Then
                MsgBox "転記先シート「" & dst_ws_name & "」が見つかりません。（" & i & "行目）", vbExclamation
                GoTo NextRow
            End If

            ' 取込元の使用範囲を起点セルから計算
            ' 起点セルが左上、CurrentRegion で囲まれた範囲をコピー
            ' ※ CurrentRegion は空白で囲まれた範囲なので、
            '   明示的に UsedRange との交差で範囲を絞る
            Dim src_range As Range
            Set src_range = ws_src.Range(sc).CurrentRegion

            ' 転記先に値のみ貼り付け（書式は報告書側を維持）
            Dim dst_range As Range
            Set dst_range = ws_dst.Range(dst_cell_addr)

            ' 行数×列数を合わせて転記（値のみ、ループで確実に）
            Dim num_rows As Long: num_rows = src_range.Rows.Count
            Dim num_cols As Long: num_cols = src_range.Columns.Count

            Dim rr As Long, cc As Long
            For rr = 1 To num_rows
                For cc = 1 To num_cols
                    dst_range.Offset(rr - 1, cc - 1).Value = _
                        src_range.Cells(rr, cc).Value
                Next cc
            Next rr

        End Select

NextRow:
        Set ws_src = Nothing
        Set ws_dst = Nothing
    Next i

    ' 開いた外部ファイルを全て閉じる
    Dim k As Variant
    For Each k In cache.Keys
        cache(k).Close SaveChanges:=False
    Next k

    Application.ScreenUpdating = True
End Sub


' ============================================================
' 特殊ラベル（詳細データシート向け）の転記
' ============================================================
Sub HandleSpecialLabel(ws_src As Worksheet, sc As String, _
                       dir As String, wks As Long, lbl As String)
    Dim ws_d As Worksheet
    Set ws_d = ThisWorkbook.Worksheets("詳細データ")

    Select Case True

    Case InStr(lbl, "時間帯別") > 0
        Dim t_row As Long
        t_row = FindLabelRowInSheet(ws_d, "⑥ 時間帯別") + 3
        Dim tr As Long, tc As Long
        For tr = 0 To 9
            For tc = 0 To 4
                ws_d.Cells(t_row + tr, 2 + tc).Value = _
                    IIf(dir = "縦", ws_src.Range(sc).Offset(tr * 5 + tc, 0).Value, _
                                    ws_src.Range(sc).Offset(tr, tc).Value)
            Next tc
        Next tr

    Case InStr(lbl, "カテゴリ別_今週") > 0
        Dim c5_row As Long
        c5_row = FindLabelRowInSheet(ws_d, "⑤ カテゴリ別") + 3
        Dim ci As Long
        For ci = 0 To 11
            ws_d.Cells(c5_row + ci, 2).Value = _
                IIf(dir = "縦", ws_src.Range(sc).Offset(ci, 0).Value, _
                                ws_src.Range(sc).Offset(0, ci).Value)
        Next ci

    Case InStr(lbl, "カテゴリ別_先週") > 0
        Dim c5r2 As Long
        c5r2 = FindLabelRowInSheet(ws_d, "⑤ カテゴリ別") + 3
        Dim ci2 As Long
        For ci2 = 0 To 11
            ws_d.Cells(c5r2 + ci2, 3).Value = _
                IIf(dir = "縦", ws_src.Range(sc).Offset(ci2, 0).Value, _
                                ws_src.Range(sc).Offset(0, ci2).Value)
        Next ci2

    Case InStr(lbl, "区局別") > 0
        Dim w_row As Long
        w_row = FindLabelRowInSheet(ws_d, "⑦ 問い合わせ") + 3
        Dim wr As Long, wc As Long
        For wr = 0 To 19
            For wc = 0 To 4
                ws_d.Cells(w_row + wr, 3 + wc).Value = _
                    IIf(dir = "縦", ws_src.Range(sc).Offset(wr * 5 + wc, 0).Value, _
                                    ws_src.Range(sc).Offset(wr, wc).Value)
            Next wc
        Next wr

    End Select
End Sub


' ============================================================
' 同フォルダからキーワードを含むExcelファイルを検索
' ============================================================
Function FindFile(keyword As String, folder As String) As String
    Dim f As String, ext As Variant
    For Each ext In Array("*.xlsx", "*.xlsm")
        f = Dir(folder & "\" & ext)
        Do While f <> ""
            If InStr(f, keyword) > 0 And f <> ThisWorkbook.Name Then
                FindFile = folder & "\" & f: Exit Function
            End If
            f = Dir()
        Loop
    Next ext
    FindFile = ""
End Function


' ============================================================
' ラベル検索ユーティリティ
' ============================================================
Function FindLabelRow(label As String) As Long
    FindLabelRow = FindLabelRowInSheet(ThisWorkbook.Worksheets("報告書"), label)
End Function

Function FindLabelRowInSheet(ws As Worksheet, label As String) As Long
    Dim r As Long
    For r = 1 To 300
        If InStr(CStr(ws.Cells(r, 1).Value), label) > 0 Then
            FindLabelRowInSheet = r: Exit Function
        End If
    Next r
    FindLabelRowInSheet = 0
End Function


' ============================================================
' 先週比をKPIサマリに反映
' ============================================================
Sub UpdateWeeklyComparison()

    Dim ws As Worksheet
    Set ws = ThisWorkbook.Worksheets("報告書")

    Dim r_zentai As Long: r_zentai = FindLabelRow("⑤全体対応件数")
    Dim r_nyuden As Long: r_nyuden = FindLabelRow("①入電件数")
    Dim r_hassin As Long: r_hassin = FindLabelRow("④発信件数")
    Dim r_houki  As Long: r_houki  = FindLabelRow("　②放棄呼数")
    Dim r_stage1 As Long: r_stage1 = FindLabelRow("第一段階")
    Dim r_stage2 As Long: r_stage2 = FindLabelRow("第二段階")
    Dim r_kpi1   As Long: r_kpi1   = FindLabelRow("全体対応件数") + 2
    Dim r_kpi2   As Long: r_kpi2   = FindLabelRow("稼働日平均受電件数") + 2

    ' KPI1段目（全体/入電/発信/放棄呼）
    Dim items As Variant
    items = Array(Array(r_zentai,"A"), Array(r_nyuden,"D"), _
                  Array(r_hassin,"G"), Array(r_houki,"J"))
    Dim it As Variant
    For Each it In items
        If it(0) > 0 Then
            Dim n As Long: n = WorksheetFunction.Sum(ws.Range("R" & it(0) & ":V" & it(0)))
            Dim p As Long: p = WorksheetFunction.Sum(ws.Range("M" & it(0) & ":Q" & it(0)))
            ws.Range(it(1) & r_kpi1).Value = FormatDiff(n - p) & "件"
        End If
    Next it

    ' 呼損率
    If r_nyuden > 0 And r_houki > 0 Then
        Dim nn As Long: nn = WorksheetFunction.Sum(ws.Range("R" & r_nyuden & ":V" & r_nyuden))
        Dim np As Long: np = WorksheetFunction.Sum(ws.Range("M" & r_nyuden & ":Q" & r_nyuden))
        Dim hn As Long: hn = WorksheetFunction.Sum(ws.Range("R" & r_houki & ":V" & r_houki))
        Dim hp As Long: hp = WorksheetFunction.Sum(ws.Range("M" & r_houki & ":Q" & r_houki))
        If nn > 0 And np > 0 Then
            ws.Range("M" & r_kpi1).Value = _
                FormatDiff(Format((hn / nn - hp / np) * 100, "0.0")) & "%"
        End If
    End If

    ' KPI2段目（第一・第二段階）
    Dim sitems As Variant
    sitems = Array(Array(r_stage1,"E"), Array(r_stage2,"I"))
    For Each it In sitems
        If it(0) > 0 Then
            Dim sn As Long: sn = WorksheetFunction.Sum(ws.Range("R" & it(0) & ":V" & it(0)))
            Dim sp As Long: sp = WorksheetFunction.Sum(ws.Range("M" & it(0) & ":Q" & it(0)))
            ws.Range(it(1) & r_kpi2).Value = FormatDiff(sn - sp) & "件"
        End If
    Next it
End Sub


' ============================================================
' 差分フォーマット
' ============================================================
Function FormatDiff(diff As Variant) As String
    Dim n As Double: n = CDbl(diff)
    If n > 0 Then
        FormatDiff = "先週比：▲" & diff
    ElseIf n < 0 Then
        FormatDiff = "先週比：▼" & Abs(diff)
    Else
        FormatDiff = "先週比：±0"
    End If
End Function


' ============================================================
' ボタン②　提出用に整える
' ============================================================
Sub PrepareForSubmission()

    If MsgBox("操作パネルと作業データシートを削除して提出用ファイルにします。" & Chr(10) & _
              "この操作は元に戻せません。よろしいですか？", _
              vbYesNo + vbExclamation, "提出用に整える") = vbNo Then Exit Sub

    Application.DisplayAlerts = False
    Dim nm As Variant
    For Each nm In Array("操作パネル", "作業データ")
        On Error Resume Next
        ThisWorkbook.Worksheets(nm).Delete
        On Error GoTo 0
    Next nm
    Application.DisplayAlerts = True

    ThisWorkbook.Worksheets("報告書").Activate
    ThisWorkbook.Save
    MsgBox "完了しました。", vbInformation
End Sub
