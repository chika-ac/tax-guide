Sub ExportCSV()
    Dim ws As Worksheet
    Dim folderPath As Variant
    Dim fNum As Integer
    Dim lastRow As Long
    Dim i As Long
    Dim j As Integer
    Dim line As String
    Dim val As String
    Dim fileNum As Integer
    Dim startRow As Long
    Dim endRow As Long
    Dim rowsPerFile As Integer
    Dim totalFiles As Integer
    Dim filePath As String

    Set ws = ThisWorkbook.Sheets("Split")

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    If lastRow < 3 Then
        MsgBox "Error: No data in Split sheet. Please run split first."
        Exit Sub
    End If

    ' 出力先フォルダを選択
    folderPath = Application.FileDialog(4).Show
    If folderPath = False Then Exit Sub
    Dim fd As Office.FileDialog
    Set fd = Application.FileDialog(4)
    fd.Title = "CSV output folder"
    If fd.Show = False Then Exit Sub
    folderPath = fd.SelectedItems(1)

    rowsPerFile = 100
    totalFiles = Int((lastRow - 2 + rowsPerFile - 1) / rowsPerFile)
    fileNum = 1

    For startRow = 3 To lastRow Step rowsPerFile
        endRow = startRow + rowsPerFile - 1
        If endRow > lastRow Then endRow = lastRow

        filePath = folderPath & "\" & "FAQ_" & Format(Now, "YYYYMMDD") & "_" & fileNum & ".csv"

        fNum = FreeFile
        Open filePath For Output As #fNum

        ' ヘッダー行
        Print #fNum, "ID,Category,PairNo,Role1,Text1,Role2,Text2"

        For i = startRow To endRow
            line = ""
            For j = 1 To 7
                val = CStr(ws.Cells(i, j).Value)
                val = Replace(val, Chr(34), Chr(34) & Chr(34))
                If InStr(val, Chr(44)) > 0 Or InStr(val, Chr(10)) > 0 Or InStr(val, Chr(34)) > 0 Then
                    val = Chr(34) & val & Chr(34)
                End If
                If j = 1 Then
                    line = val
                Else
                    line = line & Chr(44) & val
                End If
            Next j
            Print #fNum, line
        Next i

        Close #fNum
        fileNum = fileNum + 1
    Next startRow

    MsgBox totalFiles & " files wo " & folderPath & " ni hozon shimashita."
End Sub
