' ============================================================
' CSV取り込み
' ============================================================
Sub CSVを取り込む()
    Dim filePath As String
    filePath = Application.GetOpenFilename("CSVファイル,*.csv", , "CSVファイルを選択してください")
    If filePath = "False" Then Exit Sub

    Application.ScreenUpdating = False

    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("データ")

    ' 既存データをクリア（ヘッダー2行は残す）
    ws.Rows("3:" & ws.Rows.Count).ClearContents

    ' CSVを開いて貼り付け
    Dim csvWB As Workbook
    Set csvWB = Workbooks.Open(filePath, Local:=True)

    Dim csvWS As Worksheet
    Set csvWS = csvWB.Sheets(1)

    Dim lastRow As Long
    lastRow = csvWS.Cells(csvWS.Rows.Count, 1).End(xlUp).Row

    If lastRow >= 1 Then
        csvWS.Range("A1:H" & lastRow).Copy
        ws.Cells(3, 1).PasteSpecial xlPasteValues
    End If

    csvWB.Close SaveChanges:=False
    Application.CutCopyMode = False
    Application.ScreenUpdating = True

    MsgBox lastRow & "行を取り込みました", vbInformation, "CSV取り込み"
End Sub
