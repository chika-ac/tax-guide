Set copyRange = srcWs.Range(srcWs.Cells(2, 1), srcWs.Cells(lastRow, lastCol))

Dim rRow As Long
Dim rCol As Long
For rRow = 1 To copyRange.Rows.Count
    For rCol = 1 To copyRange.Columns.Count
        With destWs.Cells(2 + rRow, rCol)
            .NumberFormat = "General"
            .Value = copyRange.Cells(rRow, rCol).Text
        End With
    Next rCol
Next rRow
