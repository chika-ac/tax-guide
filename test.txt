Sub LoadFuyoData()
    Dim wb As Workbook
    Dim wbSrc As Workbook
    Dim wsSrc As Worksheet
    Dim wsDst As Worksheet
    Dim folderPath As String
    Dim fileName As String
    Dim srcLastRow As Long
    Dim srcLastCol As Long
    Dim ans As Integer
    
    ' このファイルと同じフォルダを取得
    folderPath = ThisWorkbook.Path & "\"
    
    ' 「扶養検索」で始まるファイルを検索
    fileName = Dir(folderPath & "扶養検索*.xlsx")
    
    If fileName = "" Then
        fileName = Dir(folderPath & "扶養検索*.xls")
    End If
    
    If fileName = "" Then
        MsgBox "「扶養検索」で始まるファイルが見つかりませんでした。" & vbCrLf & _
               "同じフォルダに「扶養検索_YYYYMMDD.xlsx」を置いてください。", _
               vbExclamation, "ファイル未検出"
        Exit Sub
    End If
    
    ' 複数ファイルある場合は警告
    If Dir() <> "" Then
        ans = MsgBox("「扶養検索」で始まるファイルが複数見つかりました。" & vbCrLf & _
                     "最初に見つかった「" & fileName & "」を使用します。" & vbCrLf & _
                     "続けますか？", vbYesNo + vbQuestion, "複数ファイル検出")
        If ans = vbNo Then Exit Sub
    End If
    
    ' 確認ダイアログ
    ans = MsgBox("「" & fileName & "」を読み込みます。" & vbCrLf & _
                 "元データシートの既存データは上書きされます。よろしいですか？", _
                 vbYesNo + vbQuestion, "データ読込確認")
    If ans = vbNo Then Exit Sub
    
    ' 読込元ファイルを非表示で開く
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    Set wbSrc = Workbooks.Open(folderPath & fileName, ReadOnly:=True)
    Set wsSrc = wbSrc.Sheets(1)
    Set wsDst = ThisWorkbook.Sheets("元データ")
    
    ' 読込元のデータ範囲を取得
    srcLastRow = wsSrc.Cells(wsSrc.Rows.Count, 1).End(xlUp).Row
    srcLastCol = wsSrc.Cells(1, wsSrc.Columns.Count).End(xlToLeft).Column
    
    ' 元データシートの既存データをクリア（ヘッダー除く・作業列除く）
    If wsDst.Cells(wsDst.Rows.Count, 1).End(xlUp).Row >= 2 Then
        wsDst.Range(wsDst.Cells(2, 1), wsDst.Cells(wsDst.Rows.Count, 37)).ClearContents
    End If
    
    ' データをコピー（値のみ・2行目以降）
    If srcLastRow >= 2 Then
        wsSrc.Range(wsSrc.Cells(2, 1), wsSrc.Cells(srcLastRow, srcLastCol)).Copy
        wsDst.Cells(2, 1).PasteSpecial Paste:=xlPasteValues
        Application.CutCopyMode = False
    End If
    
    ' 読込元を閉じる
    wbSrc.Close SaveChanges:=False
    
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    
    MsgBox "読込完了しました。" & vbCrLf & _
           "ファイル：" & fileName & vbCrLf & _
           srcLastRow - 1 & " 行を取り込みました。", _
           vbInformation, "読込完了"
End Sub
